e1000e 驱动是协议栈与硬件之间的"翻译官":把 `net_device_ops` 的四个关键回调翻译成对 82574/I219 系列 MAC 控制器寄存器和 DMA 描述符环的操作。本文按 probe → open → 收 → 发 → 中断的顺序读完这层;硬件寄存器语义见 [006](006.硬件层-MAC-PHY-PCIe.md)。

**环 = 描述符数组 + 两个游标 + 两个硬件寄存器游标**,这是所有 Intel 网卡驱动的通用骨架(igb/ixgbe 同款,只是环变多):

```c
// drivers/net/ethernet/intel/e1000e/hw.h:307
struct e1000_tx_desc {
	__le64 buffer_addr;      /* Address of the descriptor's data buffer */
	union {
		__le32 data;
		struct {
			__le16 length;    /* Data buffer length */
			u8 cso;	/* Checksum offset */
			u8 cmd;	/* Descriptor control */
		} flags;
	} lower;
	union {
		__le32 data;
		struct {
			u8 status;     /* Descriptor status */   // 硬件写回! DD 位在此
			u8 css;	/* Checksum start */
			__le16 special;
		} fields;
	} upper;
};
// hw.h:246 另有 union e1000_rx_desc_extended(RX 描述符, 同为 16 字节)
```

描述符是一个 **16 字节的 CPU↔硬件通信单元**:驱动往里写"数据缓冲区地址 + 命令"(lower.cmd:E1000_TXD_CMD_EOP/RS,defines.h:270/273),硬件干完活往里写"状态"(upper.status:E1000_TXD_STAT_DD/RXD_STAT_DD,defines.h:278/53)。

## 4. 发送回调:e1000_xmit_frame(netdev.c:5803)

协议栈每发一个 skb 调用一次(上游见 [008 §3](008.发包全链路-从sk_buff到网线.md)),五个步骤:

```c
// drivers/net/ethernet/intel/e1000e/netdev.c:5803(节选, 行号实测)
static netdev_tx_t e1000_xmit_frame(struct sk_buff *skb, struct net_device *netdev)
{
	...
	...   // 1) 描述符余量检查: count+2 的安全间隙, 不够让协议栈重试
	if (e1000_maybe_stop_tx(tx_ring, count + 2))
		return NETDEV_TX_BUSY;                        // :5877

	...
	first = tx_ring->next_to_use;                     // :5886 本包第一个描述符
	tso = e1000_tso(tx_ring, skb, protocol);          // :5888 TSO 卸载(分段下放硬件)
	if (tso) tx_flags |= E1000_TX_FLAGS_TSO;
	else if (e1000_tx_csum(tx_ring, skb, protocol))   // :5896 校验和卸载
		tx_flags |= E1000_TX_FLAGS_CSUM;
	...
	...   // 2) DMA 映射 + 切描述符: 线性区 + 分页区逐片映射
	count = e1000_tx_map(tx_ring, skb, first, adapter->tx_fifo_limit,
			     nr_frags);   // :5910 → e1000_tx_map :5573
	...
	skb_tx_timestamp(skb);                            // :5926 硬件时间戳钩子
	netdev_sent_queue(netdev, skb->len);              // :5928 BQL 记账
	e1000_tx_queue(tx_ring, tx_flags, count);         // :5929 → :5667, 填描述符
	...
	...   // 3) 门铃: 只在批量末尾/队列停时敲, xmit_more 批量省写
	if (!netdev_xmit_more() ||
	    netif_xmit_stopped(netdev_get_tx_queue(netdev, 0))) {   // :5936
		if (adapter->flags2 & FLAG2_PCIM2PCI_ARBITER_WA)
			e1000e_update_tdt_wa(tx_ring, tx_ring->next_to_use);  // :5939 ME 固件 workaround 版
		else
			writel(tx_ring->next_to_use, tx_ring->tail);          // :5942 ★ TDT 门铃!
	}
	...
}
```

`e1000_tx_map`(:5573)对 skb 的线性区(`dma_map_single`,:5593)与每个 frag(`dma_map_page`)把**虚拟地址换成总线地址**写进 `buffer_addr`——硬件 DMA 只认总线地址。`e1000_tx_queue`(:5667)填 `lower.flags.cmd`(EOP/RS 命令位):

```c
// netdev.c:5667(节选, e1000_tx_queue, 行号实测)
	...
	i = tx_ring->next_to_use;                         // :5702

	do {
		buffer_info = &tx_ring->buffer_info[i];
		tx_desc = E1000_TX_DESC(*tx_ring, i);         // :5706 取第 i 个描述符
		tx_desc->buffer_addr = cpu_to_le64(buffer_info->dma);   // :5707 总线地址
		tx_desc->lower.data = cpu_to_le32(txd_lower |
						  buffer_info->length);   // :5708 命令位+长度
		tx_desc->upper.data = cpu_to_le32(txd_upper);  // :5709 状态区清零
		i++; if (i == tx_ring->count) i = 0;           // 环绕
	} while (--count > 0);

	tx_desc->lower.data |= cpu_to_le32(adapter->txd_cmd);   // 末描述符补 EOP/RS
	...
	wmb();                                            // :5728 ★ 先保证描述符对硬件可见
	tx_ring->next_to_use = i;                         // :5730 推进软件游标
```

**内存屏障是驱动层的硬约束**:`wmb()`(:5728)确保描述符先于门铃落内存;门铃 `writel`(:5942)触发 PCIe Memory Write,硬件读到 TDT 才开始搬数据。`netdev_xmit_more()`(:5936,批量发送优化)把"每包一敲"改成"攒一批再敲"——QEMU 下 e1000e 同样适用。

## 5. 接收:NAPI 轮询回调 e1000e_poll(netdev.c:2656)

```c
// drivers/net/ethernet/intel/e1000e/netdev.c:2656(节选, 行号实测)
static int e1000e_poll(struct napi_struct *napi, int budget)
{
	struct e1000_adapter *adapter = container_of(napi, struct e1000_adapter, napi);
	...
	if (!adapter->msix_entries ||
	    (adapter->rx_ring->ims_val & adapter->tx_ring->ims_val))
		tx_cleaned = e1000_clean_tx_irq(adapter->tx_ring);   // :2668 先回收发送

	adapter->clean_rx(adapter->rx_ring, &work_done, budget); // :2670 再收包(e1000_clean_rx_irq :915)

	if (!tx_cleaned || work_done == budget)
		return budget;                                   // :2673 干不完, 留在轮询模式

	if (likely(napi_complete_done(napi, work_done))) {       // :2678 干完了, 退出轮询
		if (adapter->itr_setting & 3)
			e1000_set_itr(adapter);                      // 中断合并率调整
		if (!test_bit(__E1000_DOWN, &adapter->state)) {
			if (adapter->msix_entries)
				ew32(IMS, adapter->rx_ring->ims_val);    // 重新打开 RX 中断
			else
				e1000_irq_enable(adapter);               // :2685 → :2234
		}
	}

	return work_done;
}
```

## 6. 中断:ICR 读一次就关门

e1000e 的三个 ISR 家族(legacy e1000_intr :1817 / MSI e1000_intr_msi :1751 / MSI-X e1000_intr_msix :7083)结构相同,**读 ICR 即自动屏蔽本源中断**——这是 NAPI 的物理前提:

```c
// netdev.c:1751(节选, e1000_intr_msi, 行号实测)
static irqreturn_t e1000_intr_msi(int __always_unused irq, void *data)
{
	struct net_device *netdev = data;
	struct e1000_adapter *adapter = netdev_priv(netdev);
	struct e1000_hw *hw = &adapter->hw;
	u32 icr = er32(ICR);                          // :1756 ★ 读中断原因(R/clear, 读后关本源)

	if (icr & E1000_ICR_LSC) { ... }   // 链路状态变化 → watchdog(:1759 起)
	...
	...   // Disable/Fake Interrupts(注释原文): 收包后关中断进轮询
	if (napi_schedule_prep(&adapter->napi)) {           // :1801 置 SCHED 位
		adapter->total_rx_bytes += ...;
		__napi_schedule(&adapter->napi);                // :1806 挂 poll_list + raise 软中断
	}
	return IRQ_HANDLED;
}
```

寄存器访问宏([e1000.h:607/611](../../../000.SOURCE_CODE/003.LINUX-7.1.3/000.LINUX-7.1.3/drivers/net/ethernet/intel/e1000e/e1000.h#L607)):

```c
// e1000.h:607-611
#define er32(reg)	__er32(hw, E1000_##reg)          // readl(hw->hw_addr + reg 偏移)
#define ew32(reg, val)	__ew32(hw, E1000_##reg, (val))   // writel(val, hw->hw_addr + reg 偏移)
```

`E1000_ICR`/`E1000_IMS`/`E1000_IMC` 等偏移定义在 [regs.h:32/35/36](../../../000.SOURCE_CODE/003.LINUX-7.1.3/000.LINUX-7.1.3/drivers/net/ethernet/intel/e1000e/regs.h#L32);`er32`/`ew32` 底下就是 `readl/writel`——**驱动层与硬件层的全部接触面,就是这几个 MMIO 读写**。watchdog(`e1000_watchdog_task`,:5208)负责链路检测、环水位检查与 tx hang 恢复。

## 7. 小结:驱动层回调清单

```bash
ethtool -S eth0 | grep -E 'rx_packets|tx_packets|rx_missed|tx_drop'

# 3. 驱动函数现场(bpftrace)
sudo bpftrace -e '
  kprobe:e1000_xmit_frame  { printf("xmit len=%d\n", ((struct sk_buff*)arg0)->len); }
  kprobe:e1000e_poll       { @poll++; }
  kprobe:e1000_clean_rx_irq{ @clean_rx++; }
  kprobe:e1000_intr_msi    { @intr++; }
  interval:s:1 { printf("poll=%lu rx=%lu intr=%lu\n", @poll, @clean_rx, @intr); }'
```
