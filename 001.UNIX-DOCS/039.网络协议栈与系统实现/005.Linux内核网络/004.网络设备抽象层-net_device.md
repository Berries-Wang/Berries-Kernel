## 1. struct net_device:所有网卡的"身份证"

IP 层只知道"包从 `skb->dev` 出去",从不关心这块卡是 e1000e、virtio 还是 veth。统一靠 2000+ 行的大结构([netdevice.h:2124](../../../000.SOURCE_CODE/003.LINUX-7.1.3/000.LINUX-7.1.3/include/linux/netdevice.h#L2124)),与 e1000e 相关的关键字段(行号为 7.1.3 实测):

```c
// include/linux/netdevice.h:2124(节选, 字段行号为 7.1.3 实测)
struct net_device {
	struct_group(priv_flags_fast,
		unsigned long		priv_flags:32;      // :2133 内部标志
	);
	const struct net_device_ops *netdev_ops;    // :2137 ★ 驱动回调函数集
	unsigned int		real_num_tx_queues;  // :2141 实际发送队列数(e1000e=1)
	unsigned int		mtu;                 // :2151 最大传输单元
	...
	unsigned int		real_num_rx_queues;  // :2185
	char			name[IFNAMSIZ];      // :2200 "eth0"(可改)
	unsigned long		mem_end;             // :2207 寄存器内存窗结束(ISA 时代遗产,PCI 卡不用)
	unsigned long		mem_start;           // :2208
	unsigned long		base_addr;           // :2209 I/O 基址(同上)
	...
	const struct ethtool_ops *ethtool_ops;      // :2263 ethtool 回调
	...
	int			irq;                     // :2296 中断号(INTx 用; MSI/MSI-X 时为向量)
	...
	const unsigned char	*dev_addr;           // :2359 MAC 地址指针(指向下面的数组)
	unsigned int		num_rx_queues;       // :2361 分配的接收队列数
	...
	unsigned int		num_tx_queues;       // :2382
	struct Qdisc __rcu	*qdisc;              // :2383 默认排队规则
	...
};
```

> **`priv_flags`/`mem_start` 们的年代感**:`base_addr`/`irq` 字段是 ISA/PCI 通用化的化石,真正的 PCI 卡由驱动把 MMIO 地址藏在私有结构里(e1000e 放 `adapter->hw.hw_addr`),这些字段大多为 0。读源码要分清"通用接口"与"实际使用者"。

## 2. net_device_ops:驱动填的"操作函数集"

```c
// include/linux/netdevice.h:1436(节选)
struct net_device_ops {
	int			(*ndo_open)(struct net_device *dev);        // :1439
	int			(*ndo_stop)(struct net_device *dev);        // :1440
	netdev_tx_t		(*ndo_start_xmit)(struct sk_buff *skb,     // :1443(ndo_start_xmit 文档注释 :1090)
					  struct net_device *dev);
	...
	void			(*ndo_set_rx_mode)(struct net_device *dev);
	int			(*ndo_set_mac_address)(struct net_device *dev, void *addr);
	int			(*ndo_change_mtu)(struct net_device *dev, int new_mtu);
	...
	void			(*ndo_tx_timeout)(struct net_device *dev, unsigned int txqueue);
	...
};
```

e1000e 的答卷([netdev.c:7334](../../../000.SOURCE_CODE/003.LINUX-7.1.3/000.LINUX-7.1.3/drivers/net/ethernet/intel/e1000e/netdev.c#L7334),全文 24 项):

```c
// drivers/net/ethernet/intel/e1000e/netdev.c:7334
static const struct net_device_ops e1000e_netdev_ops = {
	.ndo_open		= e1000e_open,        // :4624  ip link set up → 此处
	.ndo_stop		= e1000e_close,       // :4740
	.ndo_start_xmit		= e1000_xmit_frame,   // :5803  ★ 协议栈发包唯一入口
	.ndo_get_stats64	= e1000e_get_stats64, // :5994
	.ndo_set_rx_mode	= e1000e_set_rx_mode,
	.ndo_set_mac_address	= e1000_set_mac,      // :4796
	.ndo_change_mtu		= e1000_change_mtu,   // :6040
	.ndo_eth_ioctl		= e1000_ioctl,
	.ndo_tx_timeout		= e1000_tx_timeout,
	.ndo_validate_addr	= eth_validate_addr,
	.ndo_vlan_rx_add_vid	= e1000_vlan_rx_add_vid,
	...
};
```

**这张表就是抽象层的全部意义**:协议栈调 `netdev_start_xmit` 时不 `if (驱动是 e1000e)`,只查表回调。

## 3. 生命周期:alloc → register → open

以 e1000e 为例,一块网卡从被发现到可用的三级火箭:

<pre>
PCI 枚举发现设备(probe, netdev.c:7369)            ── 详见 006
  ├─ alloc_etherdev(sizeof(struct e1000_adapter))   etherdevice.h:59 → alloc_netdev_mqs(netdevice.h:4980)
  │    └─ netdev_priv(netdev) 拿到藏在其后的私有结构   netdevice.h:2785
  ├─ netdev->netdev_ops = &e1000e_netdev_ops;        netdev.c:7460
  └─ register_netdev(netdev)                         dev.c:11531
       └─ register_netdevice :11304  挂进全局链表/命名空间, 名字"eth0"生效

ip link set eth0 up(或 ifup)
  └─ dev_open(dev) → __dev_open(dev.c:1667)
       ├─ ndo_open = e1000e_open(netdev.c:4624)      ★ 驱动侧: 申请中断/分配环/使能硬件
       │    (开头先 netif_carrier_off :4638 —— 此刻还没有链路)
       └─ netif_set_up + dev_activate :1709-1711     置 UP / 挂默认 qdisc
            … carrier_on 不在 open 路径: PHY 链路经 LSC 中断 → watchdog 上报(006 §4)

ip link set eth0 down
  └─ dev_close → __dev_close_many(dev.c:1736) → ndo_stop = e1000e_close(:4740)
</pre>

`e1000e_open` 驱动侧干了什么(节选,详见 [005](005.网卡驱动层-e1000e.md)):

```c
// drivers/net/ethernet/intel/e1000e/netdev.c:4624(节选)
int e1000e_open(struct net_device *netdev)
{
	struct e1000_adapter *adapter = netdev_priv(netdev);
	...
	err = e1000e_setup_tx_resources(adapter->tx_ring);   // :4642 dma_alloc_coherent 分配 TX 描述符环
	...
	err = e1000e_setup_rx_resources(adapter->rx_ring);   // :4647 同上(RX 环)
	...
	e1000_configure(adapter);                            // :4673 → :3754, 写 RDBA/RCTL/TCTL 装填硬件

	err = e1000_request_irq(adapter);                    // :4675 → :2153, 注册中断(MSI-X→MSI→INTx 降级)
	...
	netif_napi_set_irq(&adapter->napi, irq);             // :4699 NAPI 绑定中断号(napi_id 之源)
	napi_enable(&adapter->napi);                         // :4700 → dev.c:7680
	...
	e1000_irq_enable(adapter);                           // :4704 → :2234, ew32(IMS,…) 打开中断掩码
	...
	hw->mac.get_link_status = true;
	e1000e_trigger_lsc(adapter);                         // :4711 请求链路状态中断 → watchdog 开始盯链路
	...
}
```

## 4. NAPI:"中断 + 轮询"的混合体

高频收包时逐包中断会把 CPU 打爆,NAPI 的思路:**来包先关中断,进轮询模式攒够一批再交协议栈**。

```c
// include/linux/netdevice.h:381(节选)
struct napi_struct {
	unsigned long		state;        // :383 SCHED/NPSCHED 等状态位
	struct list_head	poll_list;    // :390 挂到 per-CPU softnet_data.poll_list
	int			weight;           // :392 本轮配额
	int			(*poll)(struct napi_struct *, int);   // :394 驱动轮询回调
	...
	struct net_device	*dev;         // :401
	struct gro_node		gro;          // :403 GRO 聚合状态
	...
	u32			napi_id;          // :411 ★ epoll/io_uring busy-poll 的钥匙(见 009/010)
	...
};
```

调度与执行的完整循环(行号实测):

<pre>
                    硬中断(ISR, netdev.c:1751 e1000_intr_msi)
                    napi_schedule(netdevice.h:558) + __napi_schedule(dev.c:6710)
                          │  把 napi_struct 挂到本 CPU softnet_data.poll_list,
                          │  raise NET_RX_SOFTIRQ
                          ▼
      软中断 net_rx_action(dev.c:7914)
          └▶ napi_poll(napi, budget)   dev.c:7786
                │ work = napi->poll(napi, weight)   ← 驱动回调 e1000e_poll(netdev.c:2656)
                ▼
      work < budget(活干完了)
        └▶ napi_complete_done(dev.c:6771)  从 poll_list 摘下 + 重新开中断(驱动内 ew32(IMS))
      work == budget(还有活)
        └▶ 留在 poll_list, 等下一轮 softirq(防饿死进程)
</pre>

非 NAPI 路径(老驱动/回环):`netif_rx`(dev.c:5764)→ `enqueue_to_backlog`(:5373)→ per-CPU backlog 的 `process_backlog`(:6644)也是一个伪装成 NAPI 的 poll 回调——所以 7.1.3 里**所有收包最终都汇入同一个软中断调度器**。

## 5. 两个分层边界函数

设备层向上/向下各开一扇门,驱动与协议栈互不相见:

|门|函数|方向|调用者|
|-|-|-|-|
|收包门|`netif_receive_skb`(dev.c:6454)|驱动 → 协议栈|`napi_gro_receive` 之后(netdevice.h:4286)|
|发包门|`dev_queue_xmit`(netdevice.h:3416 内联 → `__dev_queue_xmit` dev.c:4766)|协议栈 → 驱动|`neigh_output` 等(见 [002 §3](002.协议栈层-IP与协议分发.md))|

`__dev_queue_xmit`(:4766)内部是"排队 + 调度"两段(详见 [008 §3](008.发包全链路-从sk_buff到网线.md)):

```c
// net/core/dev.c:4766(节选)
int __dev_queue_xmit(struct sk_buff *skb, struct net_device *sb_dev)
{
	...
	txq = netdev_core_pick_tx(dev, skb, sb_dev);   // 选发送队列(netdevice.h:2711 声明)
	q = rcu_dereference_bh(txq->qdisc);            // 该队列的 qdisc
	...
	if (q->enqueue) {
		rc = __dev_xmit_skb(skb, q, dev, txq);     // 排队 → __qdisc_run(sch_generic.c:440)
		...
	}
	...
	// 无 qdisc(noqueue, 如 lo): 不排队直接发
	dev_hard_start_xmit(skb, dev, txq, &rc);       // :3894
}
```

## 6. packet_type:收包门后的二次分发

`netif_receive_skb` → `__netif_receive_skb_core`(dev.c:5972)按**协议号**把 skb 分发给注册者。IPv4 的注册者 `ip_packet_type`(af_inet.c:1881):

```c
// net/ipv4/af_inet.c:1881
static struct packet_type ip_packet_type __read_mostly = {
	.type = cpu_to_be16(ETH_P_IP),
	.func = ip_rcv,          // → 002 §2
	.gro_receive = inet_gro_receive,     // GRO 分流
	.gro_complete = inet_gro_complete,
};
```

分发现场(dev.c,行号实测):

```c
// net/core/dev.c:5972(节选, __netif_receive_skb_core)
	...
	list_for_each_entry_rcu(ptype, &dev_net_rcu(skb->dev)->ptype_all, list) {  // :6034 抓包(pcap)
		if (unlikely(pt_prev))                                                 // :6036
			ret = deliver_skb(skb, pt_prev, orig_dev);                         // :6037
		pt_prev = ptype;
	}
	...
	rx_handler = rcu_dereference(skb->dev->rx_handler);                        // :6083 网桥/bond 钩子
	if (rx_handler) {                                                          // :6084
		if (unlikely(pt_prev)) {
			ret = deliver_skb(skb, pt_prev, orig_dev);
			pt_prev = NULL;
		}
		switch (rx_handler(&skb)) {                                            // :6089
		case RX_HANDLER_CONSUMED: ...                                          // 网桥已吃掉
		case RX_HANDLER_PASS:     ...                                          // 原样继续协议栈
		}
	}
	...
	type = skb->protocol;                                                      // eth_type_trans 填的 ETH_P_IP
	deliver_ptype_list_skb(skb, &pt_prev, orig_dev, type,                      // :6155
			       &dev_net_rcu(skb->dev)->ptype_specific);            // :6156 → ip_packet_type → ip_rcv
```

- **ptype_all**:tcpdump/pcap 抓包挂这里,拿到的是"原始帧"副本;
- **ptype_specific**:ETH_P_IP → `ip_rcv`,ETH_P_ARP → arp_rcv,ETH_P_IPV6 → ipv6_rcv;
- **rx_handler**:网桥/ bonding 在这层劫持 skb——这就是为什么桥接口抓包看到的帧和协议栈看到的不一定一样。

帧→协议号的转换由 `eth_type_trans`(eth.c:155)完成([007 §3](007.收包全链路-从网线到epoll.md) 详述)。

## 7. qdisc:设备层的排队规则

每块网卡(精确到每个 txq)有一个 `struct Qdisc`。e1000e 单队列、默认 mq+pfifo_fast:`pfifo_fast_enqueue`(sch_generic.c:757)/`pfifo_fast_dequeue`(:781)是三个优先级带的三色 FIFO;`noqueue`(lo 等)走 `noqueue_init`(:712)直发路径。发送调度循环 `__qdisc_run`(:440)→ `sch_direct_xmit`(:344)在 [008](008.发包全链路-从sk_buff到网线.md) 展开。
