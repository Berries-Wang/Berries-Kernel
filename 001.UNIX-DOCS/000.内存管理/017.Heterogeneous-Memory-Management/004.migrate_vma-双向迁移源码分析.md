# 004. 双向迁移: `migrate_vma_*` 源码分析 (`mm/migrate_device.c`)

> 源码: [mm/migrate_device.c (v6.8, 964 行)](https://git.kernel.org/pub/scm/linux/kernel/git/stable/linux.git/tree/mm/migrate_device.c?h=v6.8)
> 本文件回答: **一页数据如何在 DDR↔VRAM 之间搬家, CPU 页表/GPU 页表/rmap 如何保持一致?**

---

## 1. 驱动侧调用模板 (固定五步, 伪代码)

> 下面的 `alloc_vram_page()` / `dma_copy()` 等是**伪代码**, 代表驱动的私有实现;
> `migrate_vma_*` 三个函数才是核心 mm 的真实 API (逐行源码见下文)。

```c
int driver_migrate_to_device(struct vm_area_struct *vma,
			     unsigned long start, unsigned long end)
{
	unsigned long npages = (end - start) >> PAGE_SHIFT;
	unsigned long *src, *dst;
	struct migrate_vma args = {
		.vma             = vma,
		.start           = start,
		.end             = end,
		.src             = src,       /* 内核填: 源页收集结果 */
		.dst             = dst,       /* 驱动填: 目的页 */
		.pgmap_owner     = my_device, /* 识别自己的 device-private 页 */
		.flags           = MIGRATE_VMA_SELECT_SYSTEM, /* 只迁系统内存 */
	};

	mmap_read_lock(vma->vm_mm);

	if (migrate_vma_setup(&args))     /* ① 收集+锁定+解除映射 */
		return -EINVAL;

	for (每一页 i) {                   /* ② 驱动自己的部分 */
		if (!(src[i] & MIGRATE_PFN_MIGRATE))
			continue;                 /* 这页迁不了(被 pin 等) */
		dst[i] = migrate_pfn(page_to_pfn(alloc_vram_page()));
		dma_copy(dst_page, src_page);     /* CE/DMA 搬数据 (异步可等 fence) */
	}

	migrate_vma_pages(&args);         /* ③ 提交 struct page 元数据迁移 */

	/* ④ 驱动更新自己的 GPU 页表: 成功的项→VRAM, 失败的项回滚 */

	migrate_vma_finalize(&args);      /* ⑤ CPU 页表换上最终项, 释放源页 */

	mmap_read_unlock(vma->vm_mm);
}
```

> 注意: **核心 mm 不搬运任何数据字节**。数据由驱动用设备的 DMA 引擎搬 ——
> 因为 CPU 根本读不了 VRAM。核心 mm 只负责"地址空间记账"。

## 2. `struct migrate_vma` 与方向标志 (`include/linux/migrate.h`)

```c
/* src[]/dst[] 数组元素的编码: pfn 在高位, 标志在低位 */
#define MIGRATE_PFN_VALID	(1UL << 0)   /* 该项的 pfn 字段有效 */
#define MIGRATE_PFN_MIGRATE	(1UL << 1)   /* 这页允许/已经参与迁移 */
#define MIGRATE_PFN_WRITE	(1UL << 3)   /* 源页原来是可写的 */
#define MIGRATE_PFN_SHIFT	6            /* pfn 左移 6 位存放 */

static inline unsigned long migrate_pfn(unsigned long pfn)
{
	return (pfn << MIGRATE_PFN_SHIFT) | MIGRATE_PFN_VALID;
}

/* 迁移方向选择 */
enum migrate_vma_direction {
	MIGRATE_VMA_SELECT_SYSTEM = 1 << 0,           /* 迁系统内存中的页 */
	MIGRATE_VMA_SELECT_DEVICE_PRIVATE = 1 << 1,   /* 迁本设备的 private 页 */
	MIGRATE_VMA_SELECT_DEVICE_COHERENT = 1 << 2,  /* 迁本设备的 coherent 页 */
};

struct migrate_vma {
	struct vm_area_struct	*vma;
	unsigned long		*dst;
	unsigned long		*src;
	unsigned long		cpages;   /* 实际被换掉映射的页数 */
	unsigned long		npages;   /* 收集到的页数 */
	unsigned long		start;
	unsigned long		end;
	void			*pgmap_owner;   /* 配合 SELECT_DEVICE_* 使用 */
	unsigned long		flags;
	struct page		*fault_page;    /* CPU 缺页迁回时由 mm 填 */
};
```

两个方向的真实调用者:

```
        CPU→VRAM (驱动策略主动发起)          VRAM→CPU (CPU 缺页被动触发)
   ┌─────────────────────────────┐     ┌───────────────────────────────────┐
   │ kfd: svm_migrate_vma_to_vram│     │ CPU fault → do_swap_page()        │
   │ nouveau: nouveau_dmem_      │     │   → is_device_private_entry()     │
   │  migrate_to_ram 的逆过程    │     │   → pgmap->ops->migrate_to_ram()  │
   │ uvm: uvm_hmm_migrate_*      │     │   → nouveau_dmem_migrate_to_ram() │
   │ flags=SELECT_SYSTEM         │     │   → migrate_vma (fault_page=vmf页)│
   └─────────────────────────────┘     │     flags=SELECT_DEVICE_PRIVATE   │
                                       └───────────────────────────────────┘
```

## 3. 第①步 `migrate_vma_setup()`: 收集 → 解除映射

```c
int migrate_vma_setup(struct migrate_vma *args)
{
	long nr_pages = (args->end - args->start) >> PAGE_SHIFT;

	args->start &= PAGE_MASK;
	args->end &= PAGE_MASK;
	if (!args->vma || is_vm_hugetlb_page(args->vma) ||
	    (args->vma->vm_flags & VM_SPECIAL) || vma_is_dax(args->vma))
		return -EINVAL;                  /* 只支持普通匿名内存! */
	...  /* 各种参数检查: 区间必须在 vma 内, src/dst 非空等 */

	memset(args->src, 0, sizeof(*args->src) * nr_pages);
	args->cpages = 0;
	args->npages = 0;

	migrate_vma_collect(args);     /* 1a. 遍历页表收集源页到 src[] */

	if (args->cpages)
		migrate_vma_unmap(args);   /* 1b. 换成 migration entry + 解除映射 */

	/*
	 * At this point pages are locked and unmapped, and thus they have
	 * stable content and can safely be copied to destination memory
	 * that is allocated by the drivers.
	 */
	return 0;
}
```

### 1a. collect: 页表遍历收集 (有 mmu_notifier 包夹!)

```c
static void migrate_vma_collect(struct migrate_vma *migrate)
{
	struct mmu_notifier_range range;

	/*
	 * Note that the pgmap_owner is passed to the mmu notifier callback so
	 * that the registered device driver can skip invalidating device
	 * private page mappings that won't be migrated.
	 *  ↑ 通知回调带上 owner: 没被迁移的设备页, 驱动不用白白失效自己页表
	 */
	mmu_notifier_range_init_owner(&range, MMU_NOTIFY_MIGRATE, 0,
		migrate->vma->vm_mm, migrate->start, migrate->end,
		migrate->pgmap_owner);
	mmu_notifier_invalidate_range_start(&range);    /* ★ 先广播: 都冻结 */

	walk_page_range(migrate->vma->vm_mm, migrate->start, migrate->end,
			&migrate_vma_walk_ops, migrate);  /* 逐 PTE 收集 */

	mmu_notifier_invalidate_range_end(&range);      /* ★ 解冻 */
	migrate->end = migrate->start + (migrate->npages << PAGE_SHIFT);
}
```

collect 阶段对每个 PTE 的处理要点 (源码 `migrate_vma_collect_pmd`, 关键分支):

```c

			if (vma_is_anonymous(vma)) {
				mpfn = MIGRATE_PFN_MIGRATE;   /* 空洞也算"可迁移": */
				migrate->cpages++;            /* 驱动可直接在 VRAM 造零页 */
			}
			goto next;
		}

		if (!pte_present(pte)) {

				mpfn |= MIGRATE_PFN_WRITE;
		} else {
			... /* zero page 直接标记可迁移(目标页清零即可, 不用拷贝) */

			mpfn |= pte_write(pte) ? MIGRATE_PFN_WRITE : 0;

			folio_get(folio);   /* 拿引用 pin 住, 冻结 pte */
```

### 1b. unmap: 换 migration entry (真正的"拔页")

```c
static unsigned long migrate_device_unmap(unsigned long *src_pfns,
					  unsigned long npages,
					  struct page *fault_page)
{
	...
	for (i = 0; i < npages; i++) {
		struct page *page = migrate_pfn_to_page(src_pfns[i]);
		...
		/* ZONE_DEVICE pages are not on LRU */
		if (!is_zone_device_page(page)) {
			if (!isolate_lru_page(page)) {      /* 从 LRU 摘除失败 */
				src_pfns[i] &= ~MIGRATE_PFN_MIGRATE;  /* 放弃这页 */
				restore++;
				continue;
			}
			put_page(page);
		}

		folio = page_folio(page);
		if (folio_mapped(folio))
			try_to_migrate(folio, 0);  /* ★ 所有 CPU 页表: PTE→migration entry */

		if (page_mapped(page) ||
		    !migrate_vma_check_page(page, fault_page)) {
			... /* 还被引用(被 GUP pin 等) → 恢复映射, 放弃迁移 */
			src_pfns[i] &= ~MIGRATE_PFN_MIGRATE;
			restore++;
			continue;
		}
		unmapped++;
	}
	...
}
```

`migrate_vma_check_page()` 是防 pin 检查: `page_count` 里除 mapcount 外多出的引用
(即被 GUP pin 住) → 拒绝迁移。**这就是 GUP 与 HMM 互斥的落点**。

## 4. 第③步 `migrate_vma_pages()`: 提交元数据迁移

```c
static void __migrate_device_pages(unsigned long *src_pfns,
				unsigned long *dst_pfns, unsigned long npages,
				struct migrate_vma *migrate)
{
	for (i = 0; i < npages; i++) {
		struct page *newpage = migrate_pfn_to_page(dst_pfns[i]);
		struct page *page = migrate_pfn_to_page(src_pfns[i]);
		...
		if (!page) {
			/* 源是空洞(pte_none/zero):
			   把 VRAM 新页直接插进 CPU 页表(伪装 swap entry) */
			...
			migrate_vma_insert_page(migrate, addr, newpage,
						&src_pfns[i]);
			continue;
		}
		...
		if (is_device_private_page(newpage) ||
		    is_device_coherent_page(newpage)) {
			if (mapping) {
				/* 只支持匿名页迁往显存 (文件页/swapcache 不行) */
				if (!folio_test_anon(folio) ||
				    !folio_free_swap(folio)) {
					src_pfns[i] &= ~MIGRATE_PFN_MIGRATE;
					continue;
				}
			}
		}

		if (migrate && migrate->fault_page == page)
			r = migrate_folio_extra(..., MIGRATE_SYNC_NO_COPY, 1);
		else
			r = migrate_folio(mapping, page_folio(newpage),
					page_folio(page), MIGRATE_SYNC_NO_COPY);
		if (r != MIGRATEPAGE_SUCCESS)
			src_pfns[i] &= ~MIGRATE_PFN_MIGRATE;   /* 失败回滚标记 */
	}
}
```

- `migrate_folio(..., MIGRATE_SYNC_NO_COPY)`: 只搬 `struct page` 元数据
  (mapcount/引用/rmap/memcg/交换缓存), **不拷贝数据** —— 数据是驱动 DMA 搬的;
- 成功后: rmap 指向新页, swapcache 里源页被替换; 失败的页清掉 `MIGRATE_PFN_MIGRATE`
  → 驱动据此知道要回滚。

## 5. 第⑤步 `migrate_vma_finalize()`: 换最终 PTE + 放页

```c
void migrate_device_finalize(unsigned long *src_pfns,
			unsigned long *dst_pfns, unsigned long npages)
{
	for (i = 0; i < npages; i++) {
		...
		src = page_folio(page);
		dst = page_folio(newpage);
		remove_migration_ptes(src, dst, false);  /* migration entry → 新页 PTE */
		folio_unlock(src);

		if (is_zone_device_page(page))
			put_page(page);            /* 显存页: 直接还显存 */
		else
			putback_lru_page(page);    /* DDR 页: 回 LRU */

		if (newpage != page) {
			unlock_page(newpage);
			if (is_zone_device_page(newpage))
				put_page(newpage);
			else
				putback_lru_page(newpage);
		}
	}
}
```

## 6. CPU 缺页迁回的另一半: `migrate_to_ram` 回调

CPU 摸了"在显存里"的页 → `do_swap_page()` 发现 `is_device_private_entry()` →
调用 pgmap 注册的回调 (以 nouveau 为例, `nouveau_dmem.c`):

```c
static vm_fault_t nouveau_dmem_migrate_to_ram(struct vm_fault *vmf)
{
	struct migrate_vma args = {
		.vma		= vmf->vma,
		.start		= vmf->address,
		.end		= vmf->address + PAGE_SIZE,
		.src		= &src,
		.dst		= &dst,
		.pgmap_owner	= drm->dev,
		.fault_page	= vmf->page,     /* ★ mm 传入缺页的那个 VRAM 页 */
		.flags		= MIGRATE_VMA_SELECT_DEVICE_PRIVATE,
	};

	if (migrate_vma_setup(&args) < 0)
		return VM_FAULT_SIGBUS;
	if (!args.cpages)
		return 0;

	spage = migrate_pfn_to_page(src);
	if (!spage || !(src & MIGRATE_PFN_MIGRATE))
		goto done;

	dpage = alloc_page_vma(GFP_HIGHUSER, vmf->vma, vmf->address);  /* 新 DDR 页 */
	if (!dpage)
		goto done;

	dst = migrate_pfn(page_to_pfn(dpage));

	svmm = spage->zone_device_data;
	mutex_lock(&svmm->mutex);
	nouveau_svmm_invalidate(svmm, args.start, args.end);  /* GPU 摘映射 */
	ret = nouveau_dmem_copy_one(drm, spage, dpage, &dma_addr); /* VRAM→DDR DMA */
	mutex_unlock(&svmm->mutex);
	...
	nouveau_fence_new(&fence, dmem->migrate.chan);   /* 等 CE 拷贝完成 */
	migrate_vma_pages(&args);
	nouveau_dmem_fence_done(&fence);
	...
done:
	migrate_vma_finalize(&args);
	return ret;
}

static const struct dev_pagemap_ops nouveau_dmem_pagemap_ops = {
	.page_free		= nouveau_dmem_page_free,
	.migrate_to_ram		= nouveau_dmem_migrate_to_ram,
};
```

## 7. 完整时序图: CPU→VRAM 迁移一页

```
 驱动                mm/migrate_device.c            CPU 页表        GPU (其他进程/设备)
  │ migrate_vma_setup()      │                          │                   │
  │──────┬──────────────────►│                          │                   │
  │      │ migrate_vma_collect()                        │                   │
  │      │   mmu_notifier_invalidate_range_start() ─────┼────────────────►  │ invalidate
  │      │   walk: 收集 src[], folio_get pin            │                   │ 回调(摘映射)
  │      │ migrate_vma_unmap()                          │                   │
  │      │   isolate_lru_page + try_to_migrate() ──────►│ PTE→migration     │
  │      │   check_page: 被pin? 放弃该页                │   entry           │
  │ ◄────┴──────────────  返回, src[] 就绪              │                   │
  │ 驱动: 分配 VRAM 页, dst[i]=migrate_pfn(...)         │                   │
  │ 驱动: CE DMA 拷贝 DDR→VRAM, 等 fence                │                   │
  │ migrate_vma_pages()                                 │                   │
  │──────┬──────────────────►│ migrate_folio(NO_COPY) ─►│ 元数据迁移        │
  │ ◄────┴───────────────────│ 空洞页: insert_page ────►│ (rmap/memcg换主)  │
  │ 驱动: 更新 GPU 页表 (成功→VRAM, 失败→回滚)          │                   │
  │ migrate_vma_finalize()                              │                   │
  │──────┬──────────────────►│ remove_migration_ptes ──►│ 最终 PTE          │
  │ ◄────┴────────────────── │ put/putback 释放源页     │                   │
  ▼                          │                          ▼                 ▼
```
