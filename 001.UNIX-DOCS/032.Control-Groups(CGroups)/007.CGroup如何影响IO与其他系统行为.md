## 1. fork/exit：cgroup 的生命周期总钩子（理解其他控制器的钥匙）

所有"计数型"控制器（pids、以及 memcg 的初始记账）都挂在进程创建/销毁路径上：

```
kernel/fork.c copy_process()
  ├─ cgroup_can_fork()          kernel/cgroup/cgroup.c:6856
  │     逐子系统调 ss->can_fork()  (cgroup.c:6866)
  │     ★ 任何一个说"不" → 整个 fork 失败, 返回错误码
  ├─ ...（失败则 cgroup_cancel_fork() cgroup.c:6887 回滚）
  └─ cgroup_post_fork()         kernel/cgroup/cgroup.c:6916
        逐子系统调 ss->fork()     (cgroup.c:6986)
        子进程继承父进程的 css_set (同一 cgroup 组合)

进程退出:
  cgroup_exit() → ss->exit()    kernel/cgroup/cgroup.c:7019
  css 释放       → ss->release() kernel/cgroup/cgroup.c:7106
```

```c
/* kernel/cgroup/cgroup.c:6856 完整函数体 */
int cgroup_can_fork(struct task_struct *child, struct kernel_clone_args *kargs)
{
	struct cgroup_subsys *ss;
	int i, j, ret;

	ret = cgroup_css_set_fork(kargs);          // 定位子进程将落入的 css_set
	if (ret)
		return ret;

	do_each_subsys_mask(ss, i, have_canfork_callback) {
		ret = ss->can_fork(child, kargs->cset);   // ← 控制器否决点 (如 pids_can_fork)
		if (ret)
			goto out_revert;
	} while_each_subsys_mask();

	return 0;

out_revert:
	for_each_subsys(ss, j) {
		if (j >= i)
			break;
		if (ss->cancel_fork)
			ss->cancel_fork(child, kargs->cset);  // 已同意的逐个回滚
	}

	cgroup_css_set_put_fork(kargs);

	return ret;
}
```

## 2. pids 控制器：让 fork 返回 EAGAIN（fork 炸弹防火墙）

### 2.1 计数与限额

```c
/* kernel/cgroup/pids.c:38 */
#define PIDS_MAX (PID_MAX_LIMIT + 1ULL)     // "max" = 无限制
#define PIDS_MAX_STR "max"
```

每个 cgroup 一个 `pids_cgroup`（counter + limit），charge 沿层级上溯——和 memcg 的 `page_counter` 同一个思路：

```c
/* kernel/cgroup/pids.c:166 完整函数体 */
static int pids_try_charge(struct pids_cgroup *pids, int num, struct pids_cgroup **fail)
{
	struct pids_cgroup *p, *q;

	for (p = pids; parent_pids(p); p = parent_pids(p)) {   // 从本组加到根
		int64_t new = atomic64_add_return(num, &p->counter);
		int64_t limit = atomic64_read(&p->limit);

		/*
		 * Since new is capped to the maximum number of pid_t, if
		 * p->limit is %PIDS_MAX then we know that this test will never
		 * fail.
		 */
		if (new > limit) {                              // 任何一层超限
			*fail = p;
			goto revert;                            // 全链回滚
		}
		/*
		 * Not technically accurate if we go over limit somewhere up
		 * the hierarchy, but that's tolerable for the watermark.
		 */
		pids_update_watermark(p, new);                  // pids.peak 统计
	}

	return 0;

revert:
	for (q = pids; q != p; q = parent_pids(q))
		pids_cancel(q, num);                            // 撤销已加过的层
	pids_cancel(p, num);

	return -EAGAIN;                                     // ← fork 失败的错误码
}
```

### 2.2 挂到 fork 路径

```c
/* kernel/cgroup/pids.c:273 */
static int pids_can_fork(struct task_struct *task, struct css_set *cset)
{
	struct pids_cgroup *pids, *pids_over_limit;
	int err;

	pids = css_pids(cset->subsys[pids_cgrp_id]);
	err = pids_try_charge(pids, 1, &pids_over_limit);   // fork 前先记账
	if (err)
		pids_event(pids, pids_over_limit);          // pids.events: max 计数

	return err;                                          // -EAGAIN → copy_process 失败
}
```

进程退出时归还：

```c
/* kernel/cgroup/pids.c:293 */
static void pids_release(struct task_struct *task)
{
	struct pids_cgroup *pids = css_pids(task_css(task, pids_cgrp_id));

	pids_uncharge(pids, 1);
}
```

**用户态可见效果**：组内进程数达到 `pids.max` 后，`fork()`/`pthread_create()` 返回 `EAGAIN`（bash 报 `fork: retry: Resource temporarily unavailable`），其他组不受影响。把交互 shell 所在组设个几十的限额，就是最简单的 fork 炸弹保险丝。

接口文件（cftypes pids.c:392 起）：`pids.max`（`pids_max_write` pids.c:301）、`pids.current`（`pids_current_read` pids.c:345）、`pids.events`、`pids.peak`。

## 3. io 控制器：给块设备限速（throttle）

### 3.1 结构：blkcg / blkg —— 把 bio 和 cgroup 关联起来

- `blkcg`：块 IO 控制器的 css（block/blk-cgroup.c）；
- `blkg`（blkcg_gq）：**blkcg × 磁盘队列** 的交叉状态——限速参数、统计都挂在 blkg 上；
- 每个 bio 提交前先"认亲"：

```c
/* block/blk-cgroup.c:2164 完整函数体 */
void bio_associate_blkg(struct bio *bio)
{
	struct cgroup_subsys_state *css;

	if (blk_op_is_passthrough(bio->bi_opf))   // passthrough IO 不记账
		return;

	rcu_read_lock();

	if (bio->bi_blkg)
		css = bio_blkcg_css(bio);         // 已关联过: 复用原 css
	else
		css = blkcg_css();                // 取当前任务的 blkcg

	bio_associate_blkg_from_css(bio, css);    // :2140, 映射到目标设备的 blkg,
	                                          // 存入 bio->bi_blkg
	rcu_read_unlock();
}
```

### 3.2 io.max 写入

```bash
echo "259:0 rbps=10485760 wiops=100" > /sys/fs/cgroup/backup/io.max
```

```c
/* block/blk-throttle.c:1527 tg_set_limit() 完整函数体 (v2 io.max 的 write 处理函数,
 * cftype 定义于 :1609: { .name = "max", .write = tg_set_limit };
 * v1 的 throttle.*_device 文件走另一个入口 tg_set_conf() :1343, 语义相同但格式是 "MAJ:MIN VALUE" ) */
static ssize_t tg_set_limit(struct kernfs_open_file *of,
			    char *buf, size_t nbytes, loff_t off)
{
	struct blkcg *blkcg = css_to_blkcg(of_css(of));
	struct blkg_conf_ctx ctx;
	struct throtl_grp *tg;
	u64 v[4];
	int ret;

	blkg_conf_init(&ctx, buf);

	ret = blkg_conf_open_bdev(&ctx);
	if (ret)
		goto out_finish;

	if (!blk_throtl_activated(ctx.bdev->bd_queue)) {
		ret = blk_throtl_init(ctx.bdev->bd_disk);      // 首次使用: 按需初始化
		if (ret)
			goto out_finish;
	}

	ret = blkg_conf_prep(blkcg, &blkcg_policy_throtl, &ctx);  // 解析 "主:次号",
	if (ret)                                                  // 定位 (cgroup,磁盘) 的 blkg
		goto out_finish;

	tg = blkg_to_tg(ctx.blkg);

	tg_update_carryover(tg);

	v[0] = tg->bps[READ];
	v[1] = tg->bps[WRITE];
	v[2] = tg->iops[READ];
	v[3] = tg->iops[WRITE];

	while (true) {
		char tok[27];	/* wiops=18446744073709551616 */
		char *p;
		u64 val = U64_MAX;
		int len;

		if (sscanf(ctx.body, "%26s%n", tok, &len) != 1)
			break;
		if (tok[0] == '\0')
			break;
		ctx.body += len;

		ret = -EINVAL;
		p = tok;
		strsep(&p, "=");
		if (!p || (sscanf(p, "%llu", &val) != 1 && strcmp(p, "max")))
			goto out_finish;

		ret = -ERANGE;
		if (!val)
			goto out_finish;

		ret = 0;
		if (!strcmp(tok, "rbps"))
			v[0] = val;
		else if (!strcmp(tok, "wbps"))
			v[1] = val;
		else if (!strcmp(tok, "riops"))
			v[2] = min_t(u64, val, UINT_MAX);
		else if (!strcmp(tok, "wiops"))
			v[3] = min_t(u64, val, UINT_MAX);
		else
			goto out_finish;
	}

	tg->bps[READ] = v[0];            // ★ 限速参数落进 throtl_grp
	tg->bps[WRITE] = v[1];
	tg->iops[READ] = v[2];
	tg->iops[WRITE] = v[3];

	tg_conf_updated(tg, false);      // 唤醒排队中的 bio 重新判速
	ret = 0;
out_finish:
	blkg_conf_exit(&ctx);
	return ret ?: nbytes;
}
```

### 3.3 生效点：submit_bio 路径上的"收费站"

```
submit_bio()
  └─ bio_associate_blkg()            block/blk-cgroup.c:2164  (bio ↔ blkg)
  └─ blk_throtl_bio()                (内联判断, 需要时进入慢路径)
       └─ __blk_throtl_bio()         block/blk-throttle.c:1730
            └─ tg_within_limit()     block/blk-throttle.c:1703  逐层祖先检查限速
                 ├─ 未超: throtl_charge_*_bio() 记账后放行, 继续查父层
                 └─ 超:   bio 挂到 tg->service_queue 排队 (FIFO)
                          throtl_pending_timer_fn()  :1119
                          按 bps/iops 速率定时放行下一个 bio
```

核心判断与层级攀爬：

```c
/* block/blk-throttle.c:1730 完整函数体 */
bool __blk_throtl_bio(struct bio *bio)
{
	struct request_queue *q = bdev_get_queue(bio->bi_bdev);
	struct blkcg_gq *blkg = bio->bi_blkg;
	struct throtl_qnode *qn = NULL;
	struct throtl_grp *tg = blkg_to_tg(blkg);
	struct throtl_service_queue *sq;
	bool rw = bio_data_dir(bio);
	bool throttled = false;
	struct throtl_data *td = tg->td;

	rcu_read_lock();
	spin_lock_irq(&q->queue_lock);
	sq = &tg->service_queue;

	while (true) {
		if (tg_within_limit(tg, bio, rw)) {
			/* within limits, let's charge and dispatch directly */
			throtl_charge_iops_bio(tg, bio);      // 记账后放行

			/*
			 * We need to trim slice even when bios are not being
			 * queued otherwise it might happen that a bio is not
			 * queued for a long time and slice keeps on extending
			 * and trim is not called for a long time. Now if
			 * limits are reduced suddenly we take into account
			 * all the IO dispatched so far at new low rate and
			 * newly queued IO gets a really long dispatch time.
			 *
			 * So keep on trimming slice even if bio is not queued.
			 */
			throtl_trim_slice(tg, rw);
		} else if (bio_issue_as_root_blkg(bio)) {
			/*
			 * IOs which may cause priority inversions are
			 * dispatched directly, even if they're over limit.
			 *
			 * Charge and dispatch directly, and our throttle
			 * control algorithm is adaptive, and extra IO bytes
			 * will be throttled for paying the debt
			 */
			throtl_charge_bps_bio(tg, bio);   // 可能造成优先级反转的 IO
			throtl_charge_iops_bio(tg, bio);  // (如回收写回) 直接放行, 记欠账
		} else {
			/* if above limits, break to queue */
			break;                           // 超 → 排队
		}

		/*
		 * @bio passed through this layer without being throttled.
		 * Climb up the ladder.  If we're already at the top, it
		 * can be executed directly.
		 */
		qn = &tg->qnode_on_parent[rw];       // ← 沿 blkg 层级向上, 最紧的一层说了算
		sq = sq->parent_sq;
		tg = sq_to_tg(sq);
		if (!tg) {
			bio_set_flag(bio, BIO_BPS_THROTTLED);
			goto out_unlock;             // 到达顶层: 真正放行
		}
	}

	/* out-of-limit, queue to @tg */
	throtl_log(sq, "[%c] bio. bdisp=%llu sz=%u bps=%llu iodisp=%u iops=%u queued=%d/%d",
		   rw == READ ? 'R' : 'W',
		   tg->bytes_disp[rw], bio->bi_iter.bi_size,
		   tg_bps_limit(tg, rw),
		   tg->io_disp[rw], tg_iops_limit(tg, rw),
		   sq_queued(sq, READ), sq_queued(sq, WRITE));

	td->nr_queued[rw]++;
	throtl_add_bio_tg(bio, qn, tg);          // ★ bio 挂入该组队列, 等定时器放行
	throttled = true;

	/*
	 * Update @tg's dispatch time and force schedule dispatch if @tg
	 * was empty before @bio, or the iops queue is empty and @bio will
	 * add to.  The forced scheduling isn't likely to cause undue
	 * delay as @bio is likely to be dispatched directly if its @tg's
	 * disptime is not in the future.
	 */
	if (tg->flags & THROTL_TG_WAS_EMPTY ||
	    tg->flags & THROTL_TG_IOPS_WAS_EMPTY) {
		tg_update_disptime(tg);          // 计算下次放行时刻
		throtl_schedule_next_dispatch(tg->service_queue.parent_sq, true);
	}

out_unlock:
	spin_unlock_irq(&q->queue_lock);

	rcu_read_unlock();
	return throttled;
}
```

> 细节：`bio_issue_as_root_blkg` 特判——直接 IO 回收路径的 bio 不受限速约束，防止"限速了写回 → 内存更紧张 → 更多写回"的死循环。

### 3.4 另一类 IO 控制：iocost / iolatency 的"延迟惩罚"

throttle 是**按速率排队**；iocost/iolatency 走的是另一条路——**在返回用户态时罚任务睡**：

```
    → blkcg_maybe_throttle_current()  block/blk-cgroup.c:2011
```

```c
/* kernel/cgroup/freezer.c:152 完整函数体 */
static void cgroup_freeze_task(struct task_struct *task, bool freeze)
{
	unsigned long flags;

	/* If the task is about to die, don't bother with freezing it. */
	if (!lock_task_sighand(task, &flags))     // 拿不到 sighand: 任务将死, 放弃
		return;

	if (freeze) {
		task->jobctl |= JOBCTL_TRAP_FREEZE;   // 借用信号处理路径埋"陷阱"
		signal_wake_up(task, false);          // 唤醒它去撞陷阱
	} else {
		task->jobctl &= ~JOBCTL_TRAP_FREEZE;
		wake_up_process(task);                // 解冻: 直接唤醒
	}

	unlock_task_sighand(task, &flags);
}
```

任务被唤醒后在信号处理边界"撞上陷阱"而入睡：

```c
/* kernel/signal.c:2699 完整函数体 */
static void do_freezer_trap(void)
	__releases(&current->sighand->siglock)
{
	/*
	 * If there are other trap bits pending except JOBCTL_TRAP_FREEZE,
	 * let's make another loop to give it a chance to be handled.
	 * In any case, we'll return back.
	 */
	if ((current->jobctl & (JOBCTL_PENDING_MASK | JOBCTL_TRAP_FREEZE)) !=
	     JOBCTL_TRAP_FREEZE) {
		spin_unlock_irq(&current->sighand->siglock);
		return;                                // 有别的信号要先处理: 再绕一圈
	}

	/*
	 * Now we're sure that there is no pending fatal signal and no
	 * pending traps. Clear TIF_SIGPENDING to not get out of schedule()
	 * immediately (if there is a non-fatal signal pending), and
	 * put the task into sleep.
	 */
	__set_current_state(TASK_INTERRUPTIBLE|TASK_FREEZABLE);
	clear_thread_flag(TIF_SIGPENDING);
	spin_unlock_irq(&current->sighand->siglock);
	cgroup_enter_frozen();                     // 组内"已冻"计数 +1
	schedule();                                // ★ 睡到 unfreeze

	/*
	 * We could've been woken by task_work, run it to clear
	 * TIF_NOTIFY_SIGNAL. The caller will retry if necessary.
	 */
	clear_notify_signal();
	if (unlikely(task_work_pending(current)))
		task_work_run();
}
```

与 `SIGSTOP` 的区别：freeze 经由信号处理**边界**冻结，进程在内核态的关键区段（如持有锁的 syscall 中段）不会被冻住，能停在"干净"的位置；被冻进程在 `cgroup.events` 的 `frozen` 变 1（cgroup.c:3784）后可安全做快照/检查点（容器 checkpoint-restore 的基础）。

相关的还有 `cgroup.kill`（cgroup.c:5480 cftype，`cgroup_kill_write` cgroup.c:4187）——写 1 向整组（含后代）发 SIGKILL，比逐个 kill 干净。

## 5. 设备访问：v1 devices 控制器已死，BPF 当立

- **v1 的 devices 控制器（devices.allow/deny）在 7.1.3 已整体移除**（kernel/cgroup/cgroup-v1.c 中已无 dev_cgroup 相关代码）；
- v2 从一开始就不提供 devices 文件，设备过滤统一走 **cgroup BPF**：`BPF_PROG_TYPE_CGROUP_DEVICE` 挂在 `BPF_CGROUP_DEVICE` 挂载点，在 `may_open_dev()`/设备打开路径上给出允许/拒绝判定；

```bash
# 拒绝某容器组访问块设备 (示意)
bpftool cgroup attach /sys/fs/cgroup/foo device pinned /sys/fs/bpf/deny_dev
```

## 6. 其他控制器一览（7.1.3 现状）

| 控制器 | 文件 | 接口 | 作用 |
|---|---|---|---|
| hugetlb | mm/hugetlb_cgroup.c | hugetlb.2MB.max 等 | 限制大页用量（独立于 memcg 的 memory.max） |
| rdma | kernel/cgroup/rdma.c | rdma.max | RDMA 资源（hca_handle/hca_object）限额 |

| (v1) legacy_freezer | kernel/cgroup/legacy_freezer.c | freezer.state | v1 冻结接口，仅为兼容保留 |

## 7. 小结：cgroup 影响系统行为的四种手段

| 手段 | 例子 | 用户态观感 |
|---|---|---|
| **否决操作** | pids 超限 → fork 返回 EAGAIN；memcg 超限 → malloc 返回 NULL | 系统调用失败 |
| **排队/限流** | io.max → bio 排队；cpu.max → 任务停车 | 进程"慢"，吞吐被削平 |
| **罚睡** | memory.high、io latency | 延迟增大（尾延迟毛刺） |
| **终止进程** | memory.max → memcg OOM | 进程被 SIGKILL |

加上 cpuset（改 CPU 集合）与 freezer（停止运行），六种手段覆盖了内核对用户进程"生杀予夺"的全部常规姿势。排查线上问题时，先判断现象属于哪一行，再去找对应的 cgroup 文件与计数器——这是 [008 实战篇](./008.CGroup日常排查与实战.md)的主线。
