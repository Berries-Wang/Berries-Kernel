## 1. obj_cgroup：内核内存（kmem）的字节级账本

### 1.1 为什么需要第二层对象

页级账本（`mem_cgroup->memory`）以页为单位；而 slab 对象小到 8 字节——一个 4KB slab 页上挤着几百个不同进程的对象，页必须记到某个 memcg 名下，对象却各自归属。解法：**页记在 obj_cgroup 的"记账池"，对象按字节从 objcg 扣**。

```c
/* include/linux/memcontrol.h:184 */
struct obj_cgroup {
	struct percpu_ref refcnt;         // 引用计数 (最后一个 slab 对象释放时才销毁)
	struct mem_cgroup *memcg;         // 归属的 memcg
	atomic_t nr_charged_bytes;        // 字节级余额 (不足一页的零头)
	union {
		struct list_head list;    /* protected by objcg_lock */
		struct rcu_head rcu;
	};
	bool is_root;
};
```

charge 的换算逻辑——**字节攒够一整页才动页账本**：

```c
/* mm/memcontrol.c:3363 完整函数体 */
static int __obj_cgroup_charge(struct obj_cgroup *objcg, gfp_t gfp,
			       size_t size, size_t *remainder)
{
	size_t charge_size;
	int ret;

	charge_size = PAGE_ALIGN(size);                       // 64B → 4096B(1页)
	ret = obj_cgroup_charge_pages(objcg, gfp, charge_size >> PAGE_SHIFT);
	                                                      // ← 整页走 try_charge_memcg (006 §3 主流程)
	if (!ret)
		*remainder = charge_size - size;              // 零头 (4096-64=4032B) 还给调用者
	return ret;
}

/* mm/memcontrol.c:3374 完整函数体 (中间大段注释以 ... 缩略) */
int obj_cgroup_charge(struct obj_cgroup *objcg, gfp_t gfp, size_t size)
{
	size_t remainder;
	int ret;

	if (likely(consume_obj_stock(objcg, size)))          // ① 字节库存够 → 直接返回
		return 0;
	...
	ret = __obj_cgroup_charge(objcg, gfp, size, &remainder);  // ② 整页 charge
	if (!ret && remainder)
		refill_obj_stock(objcg, remainder, false);   // ③ 零头入字节库存, 供后续小对象抵扣
	return ret;
}

void obj_cgroup_uncharge(struct obj_cgroup *objcg, size_t size)
{
	refill_obj_stock(objcg, size, true);                 // 释放也是攒字节进库存
}
```

### 1.2 归属判定：current_obj_cgroup()

charge 归属看**当前任务**（slab 场景）或 **mm**（用户页场景，[006 §2](./006.CGroup如何影响内存分配.md)）；内核线程或 `set_active_memcg()` 临时改挂的场景（如回收替别人干活）由 per-task 的 active_memcg 覆盖。

### 1.3 组删除时的"资产重组"：reparent

cgroup.rmdir 后 memcg 结构不会立刻消失（页还在、对象还在引用）。孤儿 objcg 连同未清账被过继给父组：

```c
/* mm/memcontrol.c:271 */
static void memcg_reparent_objcgs(struct mem_cgroup *memcg)
```

它遍历所有节点，把 memcg 名下 objcg 的 `->memcg` 改指父组、LRU/统计同步迁移（`__memcg_reparent_objcgs()` :209）——这就是"删除 cgroup 后 memory.current 归零、但父组用量上升"的原因。

## 2. per-CPU stock：charge 快路径无锁的秘密

### 2.1 结构：每 CPU 缓存多个 memcg 的零钱

```c
/* mm/memcontrol.c:2005 */
#define NR_MEMCG_STOCK 7

/* mm/memcontrol.c:2007 */
struct memcg_stock_pcp {
	local_trylock_t lock;
	uint8_t nr_pages[NR_MEMCG_STOCK];      // 每个槽的页数余额
	struct mem_cgroup *cached[NR_MEMCG_STOCK];  // 槽对应的 memcg (LRU 淘汰)
	struct work_struct work;
	unsigned long flags;
	uint8_t drain_idx;
};

static DEFINE_PER_CPU_ALIGNED(struct memcg_stock_pcp, memcg_stock) = ...;
```

7.1.3 的变化：单槽变 **7 槽数组**——容器宿主机上同一 CPU 常在多个 memcg 间切换（中断打进别的容器任务），单槽会不停换人清空重灌。

### 2.2 取钱：本地 trylock

```c
/* mm/memcontrol.c:2054 完整函数体 */
static bool consume_stock(struct mem_cgroup *memcg, unsigned int nr_pages)
{
	...
	if (nr_pages > MEMCG_CHARGE_BATCH ||
	    !local_trylock(&memcg_stock.lock))        // 本地 trylock, 抢不到就走慢路径
		return ret;

	stock = this_cpu_ptr(&memcg_stock);

	for (i = 0; i < NR_MEMCG_STOCK; ++i) {
		if (memcg != READ_ONCE(stock->cached[i]))
			continue;
		stock_pages = READ_ONCE(stock->nr_pages[i]);
		if (stock_pages >= nr_pages) {
			WRITE_ONCE(stock->nr_pages[i], stock_pages - nr_pages);
			ret = true;
		}
		break;
	}
	local_unlock(&memcg_stock.lock);
	return ret;
}
```

存钱在 `try_charge_memcg()` 的 `done_restock`（memcontrol.c:2706）：一次 charge 批量拿 `MEMCG_CHARGE_BATCH = 64` 页（memcontrol.h:334），用剩的 `refill_stock()`（:2152）入槽：

```c
/* mm/memcontrol.c:2152 完整函数体 (尾部 LRU 换槽逻辑以 ... 缩略) */
static void refill_stock(struct mem_cgroup *memcg, unsigned int nr_pages)
{
	struct memcg_stock_pcp *stock;
	struct mem_cgroup *cached;
	uint8_t stock_pages;
	bool success = false;
	int empty_slot = -1;
	int i;

	/*
	 * For now limit MEMCG_CHARGE_BATCH to 127 and less. ...
	 */
	BUILD_BUG_ON(MEMCG_CHARGE_BATCH > S8_MAX);

	VM_WARN_ON_ONCE(mem_cgroup_is_root(memcg));

	if (nr_pages > MEMCG_CHARGE_BATCH ||
	    !local_trylock(&memcg_stock.lock)) {
		/*
		 * In case of larger than batch refill or unlikely failure to
		 * lock the percpu memcg_stock.lock, uncharge memcg directly.
		 */
		memcg_uncharge(memcg, nr_pages);      // 超批量: 不缓存直接退回总账
		return;
	}

	stock = this_cpu_ptr(&memcg_stock);
	for (i = 0; i < NR_MEMCG_STOCK; ++i) {
		cached = READ_ONCE(stock->cached[i]);
		if (!cached && empty_slot == -1)
			empty_slot = i;               // 记住空槽 (可能要收留新 memcg)
		if (memcg == READ_ONCE(stock->cached[i])) {
			stock_pages = READ_ONCE(stock->nr_pages[i]) + nr_pages;
			WRITE_ONCE(stock->nr_pages[i], stock_pages);
			if (stock_pages > MEMCG_CHARGE_BATCH)
				drain_stock(stock, i); // 槽满(>64页): 排空回总账
			success = true;
			break;
		}
	}
	...
	// success == false 时: 用 empty_slot 收留该 memcg
	//   (7 槽全被占用 → 淘汰 drain_idx 指向的最老槽)
	local_unlock(&memcg_stock.lock);
}
```

### 2.3 一致性兜底：drain

limit 生效、组删除、内存紧张时需要把散在各 CPU 的零钱收回总账：

```c
/* mm/memcontrol.c:2638 (try_charge_memcg 内) */
	if (!drained) {
		drain_all_stock(mem_over_limit);   // IPI 各 CPU: 排空库存回账本
		drained = true;
		goto retry;
	}
```

`memory_max_write()` 降限额时同样先 `drain_all_stock()`（memcontrol.c:4740 附近）再判断超限——否则库存里的"零钱"会绕过新限额。

kmem 侧还有一个**字节级**库存（`struct obj_stock_pcp` memcontrol.c:2021，`cached_objcg` + `nr_bytes`），与页库存同构，供 `__consume_obj_stock()`（:3212，调用点 :3494）快路径使用。

## 3. 统计聚合：memory.stat 的数字从哪来

### 3.1 记账：per-CPU 副本 + 阈值标脏

每次页状态变化（如 `mod_memcg_state`/`count_memcg_events` 的内联入口）只更新**本 CPU 副本** `memcg->vmstats_percpu`，同时累加 `stats_updates` 计数；累计到阈值（`MEMCG_CHARGE_BATCH * nr_cpus`，memcontrol.c:666 `memcg_vmstats_needs_flush()`）才通过 rstat 通道标脏一次（复用 [009 §4](./009.CGroup核心机制深挖.md) 的 `__css_rstat_updated()` rstat.c:71 机制）。

```c
/* mm/memcontrol.c:628 (节选: 省略对称的 events/local/pending 数组) */
struct memcg_vmstats {
	/* 聚合后的只读视图 (memory.stat 读这里) */
	long state[MEMCG_VMSTAT_SIZE];
	unsigned long events[NR_MEMCG_EVENTS];
	/* 本组自己的量 (memory.events.local / numa_stat) */
	long state_local[...];
	...
	/* 待上传父组的量 */
	atomic_long_t state_pending[...];
	...
};
```

### 3.2 读取：rstat flush 三视角聚合

`cat memory.stat` 触发 `css_rstat_flush()` → memcg 的回调：

```c
/* mm/memcontrol.c:4381 完整函数体 */
static void mem_cgroup_css_rstat_flush(struct cgroup_subsys_state *css, int cpu)
{
	struct mem_cgroup *memcg = mem_cgroup_from_css(css);
	struct mem_cgroup *parent = parent_mem_cgroup(memcg);
	struct memcg_vmstats_percpu *statc;
	struct aggregate_control ac;
	int nid;

	flush_nmi_stats(memcg, parent, cpu);          // 7.1.3: NMI 上下文的记账也并入

	statc = per_cpu_ptr(memcg->vmstats_percpu, cpu);

	ac = (struct aggregate_control) {
		.aggregate = memcg->vmstats->state,        // 全量视图 (含子孙)
		.local     = memcg->vmstats->state_local,  // 仅本组
		.pending   = memcg->vmstats->state_pending,// 待上传父组的增量
		.ppending  = parent ? parent->vmstats->state_pending : NULL,
		.cstat     = statc->state,                 // 本 CPU 副本
		.cstat_prev= statc->state_prev,            // 上次聚合快照
		.size      = MEMCG_VMSTAT_SIZE,
	};
	mem_cgroup_stat_aggregate(&ac);                // delta = cstat - cstat_prev,
	                                              // 累入 aggregate/local/pending

	ac = (struct aggregate_control) {              // events 同样处理 (memory.events*)
		.aggregate = memcg->vmstats->events,
		.local = memcg->vmstats->events_local,
		.pending = memcg->vmstats->events_pending,
		.ppending = parent ? parent->vmstats->events_pending : NULL,
		.cstat = statc->events,
		.cstat_prev = statc->events_prev,
		.size = NR_MEMCG_EVENTS,
	};
	mem_cgroup_stat_aggregate(&ac);

	for_each_node_state(nid, N_MEMORY) {           // 每 NUMA 节点的 lruvec 统计
		struct mem_cgroup_per_node *pn = memcg->nodeinfo[nid];
		struct lruvec_stats *lstats = pn->lruvec_stats;
		struct lruvec_stats *plstats = NULL;
		struct lruvec_stats_percpu *lstatc;

		if (parent)
			plstats = parent->nodeinfo[nid]->lruvec_stats;

		lstatc = per_cpu_ptr(pn->lruvec_stats_percpu, cpu);

		ac = (struct aggregate_control) {
			.aggregate = lstats->state,
			.local = lstats->state_local,
			.pending = lstats->state_pending,
			.ppending = plstats ? plstats->state_pending : NULL,
			.cstat = lstatc->state,
			.cstat_prev = lstatc->state_prev,
			.size = NR_MEMCG_NODE_STAT_ITEMS,
		};
		mem_cgroup_stat_aggregate(&ac);

	}
	WRITE_ONCE(statc->stats_updates, 0);           // 清零本 CPU 的"待聚合"计数
	if (atomic_long_read(&memcg->vmstats->stats_updates))
		atomic_long_set(&memcg->vmstats->stats_updates, 0);
}
```

三个视角一次算清：`memory.stat`（含子孙的层级值）、`memory.events.local`（仅本组）、父组的 pending（下一轮 flush 时上传）。7.1.3 还加了 `flush_nmi_stats()`——NMI 里也能安全记账。

**运维含义**：`memory.stat` 各行之和 ≈ `memory.current`（差异主要来自批量 flush 的窗口）；高频读大量组的 memory.stat 本身就是 CPU 开销（见 [008 §2](./008.CGroup日常排查与实战.md)）。

## 4. PSI：memory.pressure 的数字从哪来

PSI（Pressure Stall Information）在 cgroup 创建时就已挂好：

```c
/* kernel/sched/psi.c:1112 */
int psi_cgroup_alloc(struct cgroup *cgroup)     // ← cgroup_create() cgroup.c:5858 调用
```

memcg 场景的关键标注点是"任务因 memcg 卡在回收里"：

```c
/* kernel/sched/psi.c:1056 完整函数体 */
void psi_memstall_enter(unsigned long *flags)
{
	struct rq_flags rf;
	struct rq *rq;

	if (static_branch_likely(&psi_disabled))
		return;

	*flags = current->in_memstall;
	if (*flags)
		return;                            // 已在 stall 区间: 幂等
	/*
	 * in_memstall setting & accounting needs to be atomic wrt
	 * changes to the task's scheduling state, otherwise we can
	 * race with CPU migration.
	 */
	rq = this_rq_lock_irq(&rf);

	current->in_memstall = 1;
	psi_task_change(current, 0, TSK_MEMSTALL | TSK_MEMSTALL_RUNNING);
	                                       // 把当前任务标记为 MEMSTALL, PSI 的采样
	                                       // 周期里这些任务的时间被计入 memory 压力
	                                       // 的 "some/full"
	rq_unlock_irq(rq, &rf);
}
```

try_charge 的两处调用：同步回收前 `psi_memstall_enter()`（memcontrol.c:2629）、`reclaim_high()` 里同样成对出现（:2318/:2323）——**分配者在 memcg 限额下花的每一微秒回收/等待时间，都是 memory.pressure some 的来源**；全组都卡住则计入 full。读取走 `cgroup_pressure_show()`（cgroup.c:4037，`memory.pressure` cftype @5504）。

这解释了 [008 §2.3](./008.CGroup日常排查与实战.md) 的判读法：系统整体空闲但组内 `memory.pressure some` 高 → 组在自己的限额里排队，不是机器缺内存。

```c
/* mm/vmscan.c:7905 完整函数体 */
int user_proactive_reclaim(char *buf,
			   struct mem_cgroup *memcg, pg_data_t *pgdat)
{
	unsigned int nr_retries = MAX_RECLAIM_RETRIES;
	unsigned long nr_to_reclaim, nr_reclaimed = 0;
	int swappiness = -1;
	char *old_buf, *start;
	substring_t args[MAX_OPT_ARGS];
	gfp_t gfp_mask = GFP_KERNEL;

	if (!buf || (!memcg && !pgdat) || (memcg && pgdat))
		return -EINVAL;                       // 二选一: memcg 或 NUMA 节点

	buf = strstrip(buf);

	old_buf = buf;
	nr_to_reclaim = memparse(buf, &buf) / PAGE_SIZE;   // "256M" → 页数
	if (buf == old_buf)
		return -EINVAL;

	buf = strstrip(buf);

	while ((start = strsep(&buf, " ")) != NULL) {      // 可选参数解析
		if (!strlen(start))
			continue;
		switch (match_token(start, tokens, args)) {
		case MEMORY_RECLAIM_SWAPPINESS:      // swappiness=0..200
			if (match_int(&args[0], &swappiness))
				return -EINVAL;
			if (swappiness < MIN_SWAPPINESS ||
			    swappiness > MAX_SWAPPINESS)
				return -EINVAL;
			break;
		case MEMORY_RECLAIM_SWAPPINESS_MAX:  // swappiness.max → 只收匿名页
			swappiness = SWAPPINESS_ANON_ONLY;
			break;
		default:
			return -EINVAL;
		}
	}

	while (nr_reclaimed < nr_to_reclaim) {
		/* Will converge on zero, but reclaim enforces a minimum */
		unsigned int batch_size = (nr_to_reclaim - nr_reclaimed) / 4;   // 分批: 剩余量的1/4
		unsigned long reclaimed;

		if (signal_pending(current))
			return -EINTR;               // 可被 Ctrl-C 打断

		/*
		 * This is the final attempt, drain percpu lru caches in the
		 * hope of introducing more evictable pages.
		 */
		if (!nr_retries)
			lru_add_drain_all();         // 最后一轮: 冲刷 per-CPU LRU 缓存

		if (memcg) {
			unsigned int reclaim_options;

			reclaim_options = MEMCG_RECLAIM_MAY_SWAP |
					  MEMCG_RECLAIM_PROACTIVE;
			reclaimed = try_to_free_mem_cgroup_pages(memcg,
					 batch_size, gfp_mask,
					 reclaim_options,    // ← PROACTIVE 标记
					 swappiness == -1 ? NULL : &swappiness);
		} else {
			...                              // 节点级 (node_reclaim) 分支, 从略
		}

		if (!reclaimed && !nr_retries--)
			return -EAGAIN;

		nr_reclaimed += reclaimed;
	}

	return 0;
}
```

要点：分四批递归收敛（避免一次hold锁太久）；`MEMCG_RECLAIM_PROACTIVE` 让回收器跳过 memory.low/min 保护语义里的部分限制（针对性回收不施加保护，见 [006 §4](./006.CGroup如何影响内存分配.md) 的 special case 注释）；可指定 swappiness 覆盖全局倾向。这正是 systemd-oomd、数据库预热、快照前瘦身等工具的内核底座。

## 6. 小结：memcg 的"三个缓存 + 一棵树"

| 部件 | 位置 | 粒度 | 谁消费 |
|---|---|---|---|
| 页库存 `memcg_stock_pcp` | memcontrol.c:2007 | 64 页 | try_charge_memcg 快路径 |
| 字节库存 `obj_stock_pcp` | memcontrol.c:2021 | 字节 | slab 记账快路径 |
| 统计副本 `vmstats_percpu` | memcontrol.c:609 起 | 各类事件计数 | memory.stat / events（经 rstat 树聚合） |
| rstat 更新树 | rstat.c:71/:137 | css 粒度标脏 | flush 时的遍历剪枝 |

设计哲学与 CPU 侧完全同构（对照 [004 §4](./004.CGroup如何影响CPU调度.md)）：**热路径只碰本 CPU 的宽松副本 + 便宜标记，全局精确状态交给冷路径的定时器/读取点收敛**——cgroup 给内核加"资源账本"而不拖慢内核，靠的就是这一套。
