## 1. 目录与文件从哪来：kernfs 与 css 的诞生

### 1.1 mkdir 一个 cgroup

cgroupfs 建立在 **kernfs** 之上（kernfs 是专为 sysfs/cgroupfs 这类"伪文件系统"做的轻量层，每个文件/目录是一个 `kernfs_node`）。`mkdir` 直接映射到：

```c
/* kernel/cgroup/cgroup.c:5965 完整函数体 */
int cgroup_mkdir(struct kernfs_node *parent_kn, const char *name, umode_t mode)
{
	struct cgroup *parent, *cgrp;
	int ret;

	/* do not accept '\n' to prevent making /proc/<pid>/cgroup unparsable */
	if (strchr(name, '\n'))
		return -EINVAL;

	parent = cgroup_kn_lock_live(parent_kn, false);      // 活组检查 + cgroup_mutex
	if (!parent)
		return -ENODEV;

	if (!cgroup_check_hierarchy_limits(parent)) {        // :5942 深度/后代数限额
		ret = -EAGAIN;                               //   cgroup.max.depth/max.descendants
		goto out_unlock;
	}

	cgrp = cgroup_create(parent, name, mode);           // :5816 分配 struct cgroup
	if (IS_ERR(cgrp)) {
		ret = PTR_ERR(cgrp);
		goto out_unlock;
	}

	/*
	 * This extra ref will be put in css_free_rwork_fn() and guarantees
	 * that @cgrp->kn is always accessible.
	 */
	kernfs_get(cgrp->kn);

	ret = css_populate_dir(&cgrp->self);                 // 生成 cgroup.procs 等核心文件
	if (ret)
		goto out_destroy;

	ret = cgroup_apply_control_enable(cgrp);             // :3336 为"父组已启用"的每个
	if (ret)                                             //   控制器创建 css 并上线
		goto out_destroy;

	TRACE_CGROUP_PATH(mkdir, cgrp);

	/* let's create and online css's */
	kernfs_activate(cgrp->kn);                           // 目录在 kernfs 中可见

	ret = 0;
	goto out_unlock;

out_destroy:
	cgroup_destroy_locked(cgrp);
out_unlock:
	cgroup_kn_unlock(parent_kn);
	return ret;
}
```

### 1.2 css 创建：控制器的实例化

`+cpu` 写进父组的 `cgroup.subtree_control` 后（`cgroup_subtree_control_write()` cgroup.c:3504 → `cgroup_apply_control()` :3428），子树每个组都会得到该控制器的 css：

```c
/* kernel/cgroup/cgroup.c:5763 完整函数体 */
static struct cgroup_subsys_state *css_create(struct cgroup *cgrp,
					      struct cgroup_subsys *ss)
{
	struct cgroup *parent = cgroup_parent(cgrp);
	struct cgroup_subsys_state *parent_css = cgroup_css(parent, ss);
	struct cgroup_subsys_state *css;
	int err;

	lockdep_assert_held(&cgroup_mutex);

	css = ss->css_alloc(parent_css);              // ← 调控制器自己的构造函数
	                                              //   cpu: cpu_cgroup_css_alloc (core.c:9500)
	                                              //   memory: mem_cgroup_css_alloc
	if (!css)
		css = ERR_PTR(-ENOMEM);
	if (IS_ERR(css))
		return css;

	init_and_link_css(css, ss, cgrp);

	err = percpu_ref_init(&css->refcnt, css_release, 0, GFP_KERNEL);   // css 引用计数
	if (err)
		goto err_free_css;

	err = cgroup_idr_alloc(&ss->css_idr, NULL, 2, 0, GFP_KERNEL);      // css->id
	if (err < 0)
		goto err_free_css;
	css->id = err;

	err = css_rstat_init(css);                    // 挂上 rstat (统计聚合, 见 §4)
	if (err)
		goto err_free_css;

	/* @css is ready to be brought online now, make it visible */
	list_add_tail_rcu(&css->sibling, &parent_css->children);
	cgroup_idr_replace(&ss->css_idr, css, css->id);

	err = online_css(css);                        // ss->css_online(), cgroup.events: populated
	if (err)
		goto err_list_del;

	return css;
	...
}
```

**css 的一生**：`css_create`（分配）→ `online_css`（上线）→ 引用被 `percpu_ref` 管理 → 组被 rmdir 时 `kill_css` → `css_killed_work_fn()`（cgroup.c:6023，经 RCU work）→ `offline_css` → `css_free`。引用计数归零才真正释放，所以内核路径里 `rcu_read_lock()` 下摸到的 css 不会凭空消失。

### 1.3 v2 的"无内部进程"约束

写入 `cgroup.procs` 前有个前置检查：

```c
/* kernel/cgroup/cgroup.c:2761 (注释) */
 * On the default hierarchy, except for the mixable, (possible) thread root
 * and threaded cgroups, subtree_control must be zero for migration
 * destination cgroups with tasks so that child cgroups don't compete
 * against tasks.
```

即：**一个组要么有进程、要么开控制器，不能两者兼得**——否则组内任务会和子组直接竞争，层级语义就乱了。这就是为什么 v2 操作总是"建 A、A 开 subtree_control、进程放进 A 的子组"。

## 2. css_set：进程与 cgroup 的多对多身份表

### 2.1 为什么需要它

一个进程同时属于每个控制器的一个组（cpu 在 A，memory 在 B……）。如果 `task_struct` 直接挂 N 个控制器指针，fork/迁移都要逐个改；css_set 把"一组控制器组合"打包成一个对象，**相同组合的所有进程共享同一个 css_set**（同一容器的几百个线程 → 1 个 css_set）。全系统用哈希表去重：

```c
/* kernel/cgroup/cgroup.c:911 */
static DEFINE_HASHTABLE(css_set_table, CSS_SET_HASH_BITS);
```

### 2.2 attach 时如何找到/构造目标 css_set

```c
/* kernel/cgroup/cgroup.c:1171 完整函数体 */
static struct css_set *find_css_set(struct css_set *old_cset,
				    struct cgroup *cgrp)
{
	struct cgroup_subsys_state *template[CGROUP_SUBSYS_COUNT] = { };
	struct css_set *cset;
	struct list_head tmp_links;
	struct cgrp_cset_link *link;
	struct cgroup_subsys *ss;
	unsigned long key;
	int ssid;

	lockdep_assert_held(&cgroup_mutex);

	/* First see if we already have a cgroup group that matches
	 * the desired set */
	spin_lock_irq(&css_set_lock);
	cset = find_existing_css_set(old_cset, cgrp, template);   // 先查哈希表 (模板 = 旧 subsys[]
	if (cset)                                                     //         换掉本层级的目标)
		get_css_set(cset);
	spin_unlock_irq(&css_set_lock);

	if (cset)
		return cset;                                     // ★ 常见情况: 复用已有 cset, 零分配

	cset = kzalloc_obj(*cset);                           // 否则才新建
	if (!cset)
		return NULL;

	/* Allocate all the cgrp_cset_link objects that we'll need */
	if (allocate_cgrp_cset_links(cgroup_root_count, &tmp_links) < 0) {
		kfree(cset);
		return NULL;
	}

	refcount_set(&cset->refcount, 1);
	cset->dom_cset = cset;
	INIT_LIST_HEAD(&cset->tasks);
	INIT_LIST_HEAD(&cset->mg_tasks);
	INIT_LIST_HEAD(&cset->dying_tasks);
	INIT_LIST_HEAD(&cset->task_iters);
	INIT_LIST_HEAD(&cset->threaded_csets);
	INIT_HLIST_NODE(&cset->hlist);
	INIT_LIST_HEAD(&cset->cgrp_links);
	INIT_LIST_HEAD(&cset->mg_src_preload_node);
	INIT_LIST_HEAD(&cset->mg_dst_preload_node);
	INIT_LIST_HEAD(&cset->mg_node);

	/* Copy the set of subsystem state objects generated in
	 * find_existing_css_set() */
	memcpy(cset->subsys, template, sizeof(cset->subsys));

	spin_lock_irq(&css_set_lock);
	/* Add reference counts and links from the new css_set. */
	list_for_each_entry(link, &old_cset->cgrp_links, cgrp_link) {
		struct cgroup *c = link->cgrp;

		if (c->root == cgrp->root)
			c = cgrp;                               // 同层级的祖先链换成新组
		link_css_set(&tmp_links, cset, c);           // 建立 cset↔cgroup 双向链接
	}

	BUG_ON(!list_empty(&tmp_links));

	css_set_count++;

	/* Add @cset to the hash table */
	key = css_set_hash(cset->subsys);
	hash_add(css_set_table, &cset->hlist, key);          // 入表, 下一个进程就能复用

	for_each_subsys(ss, ssid) {
		struct cgroup_subsys_state *css = cset->subsys[ssid];

		list_add_tail(&cset->e_cset_node[ssid],
			      &css->cgroup->e_csets[ssid]);
		css_get(css);
	}

	spin_unlock_irq(&css_set_lock);
	...
	return cset;
}
```

### 3.2 事务主体：全或无（all-or-nothing）

```c
/* kernel/cgroup/cgroup.c:2654 完整函数体 */
static int cgroup_migrate_execute(struct cgroup_mgctx *mgctx)
{
	struct cgroup_taskset *tset = &mgctx->tset;
	struct cgroup_subsys *ss;
	struct task_struct *task, *tmp_task;
	struct css_set *cset, *tmp_cset;
	int ssid, failed_ssid, ret;

	/* check that we can legitimately attach to the cgroup */
	if (tset->nr_tasks) {
		do_each_subsys_mask(ss, ssid, mgctx->ss_mask) {
			if (ss->can_attach) {                    // ① 预检: 逐控制器问"能不能来"
				tset->ssid = ssid;
				ret = ss->can_attach(tset);      //    cpu: RT 准入检查 (004 §2)
				if (ret) {                       //    cpuset: CPU 兼容检查 (004 §5.1)
					failed_ssid = ssid;      //    pids: 配额检查 (007 §2)
					goto out_cancel_attach;
				}
			}
		} while_each_subsys_mask();
	}

	/*
	 * Now that we're guaranteed success, proceed to move all tasks to
	 * the new cgroup. There are no failure cases after here, so this
	 * is the commit point.
	 */
	spin_lock_irq(&css_set_lock);
	list_for_each_entry(cset, &tset->src_csets, mg_node) {
		list_for_each_entry_safe(task, tmp_task, &cset->mg_tasks, cg_list) {
			struct css_set *from_cset = task_css_set(task);
			struct css_set *to_cset = cset->mg_dst_cset;

			get_css_set(to_cset);
			to_cset->nr_tasks++;
			css_set_move_task(task, from_cset, to_cset, true);   // ② 提交点: 换 task->cgroups 指针
			from_cset->nr_tasks--;
			/*
			 * If the source or destination cgroup is frozen,
			 * the task might require to change its state.
			 */
			cgroup_freezer_migrate_task(task, from_cset->dfl_cgrp,
						    to_cset->dfl_cgrp);   //    冻结状态随迁 (007 §4)
			put_css_set_locked(from_cset);

		}
	}
	spin_unlock_irq(&css_set_lock);

	/*
	 * Migration is committed, all target tasks are now on dst_csets.
	 * Nothing is sensitive to fork() after this point.  Notify
	 * controllers that migration is complete.
	 */
	tset->csets = &tset->dst_csets;

	if (tset->nr_tasks) {
		do_each_subsys_mask(ss, ssid, mgctx->ss_mask) {
			if (ss->attach) {                        // ③ 事后通知: 各控制器执行真正的资源迁移
				tset->ssid = ssid;               //    cpu: sched_move_task() 换运行队列 (004 §2)
				ss->attach(tset);                //    cpuset: set_cpus_allowed_ptr
			}
		} while_each_subsys_mask();
	}

	ret = 0;
	goto out_release_tset;

out_cancel_attach:                                     // can_attach 失败: 已同意的逐个 cancel
	if (tset->nr_tasks) {
		do_each_subsys_mask(ss, ssid, mgctx->ss_mask) {
			if (ssid == failed_ssid)
				break;                           // 走到失败的这家为止
			if (ss->cancel_attach) {
				tset->ssid = ssid;
				ss->cancel_attach(tset);
			}
		} while_each_subsys_mask();
	}
out_release_tset:                                      // 善后: 任务搬回原链表, 迁移上下文复位
	spin_lock_irq(&css_set_lock);
	list_splice_init(&tset->dst_csets, &tset->src_csets);
	list_for_each_entry_safe(cset, tmp_cset, &tset->src_csets, mg_node) {
		list_splice_tail_init(&cset->mg_tasks, &cset->tasks);
		list_del_init(&cset->mg_node);
	}
	spin_unlock_irq(&css_set_lock);

	/*
	 * Re-initialize the cgroup_taskset structure in case it is reused
	 * again in another cgroup_migrate_add_task()/cgroup_migrate_execute()
	 * iteration.
	 */
	tset->nr_tasks = 0;
	tset->csets    = &tset->src_csets;
	return ret;
}
```

三个阶段值得咀嚼：
- **① can_attach 阶段**在换任何指针**之前**——所有控制器先表态，避免"cpu 同意了、cpuset 拒绝"的半迁移状态；
- **② 提交点**只做一件事：把 `task->cgroups` 从旧 css_set 换成新 css_set（新 cset 早在 prepare 阶段就构造好了），这一步在 `css_set_lock` 下是原子的——此后进程在所有控制器眼里都属于新组；
- **③ attach 阶段**做重活（调度器换队列、cpuset 改掩码、memcg 换记账对象），失败也没有回头路（can_attach 已保证不会失败）。

### 3.3 fork/exit 与 css_set

子进程复制父进程的 cset（`cgroup_fork`/`cgroup_post_fork()` cgroup.c:6916，把 child 挂到 `cset->tasks` 链表并逐控制器 `ss->fork`）；退出时 `cgroup_exit()` 逐子系统 `ss->exit()`（:7019）移出链表。迁移期间 `cgroup_threadgroup_rwsem` 把"遍历线程组"与"fork 新线程"互斥——这就是 can_attach 能安全地对"整个线程组"做检查的原因。

## 4. rstat：cpu.stat / memory.stat 的数字从哪来

### 4.1 问题：per-CPU 记账 vs 层级读数

如果每次 tick 都更新全局计数器，上千个 CPU 同时记账会 cacheline 弹跳到死；但 cgroup.stat 又是全系统一棵树。解法是 **rstat（hierarchical relay statistics）**：记账只动 per-CPU 副本并"标脏"，读取时再逐级聚合。

### 4.2 记账：标脏（几乎无锁）

以 cpu.stat 的 usage_usec 为例，调度器每次 `update_curr()` 都会记账（fair.c:1377 `cgroup_account_cputime(donor, delta_exec)`），内联快速路径后进入：

```c
/* kernel/cgroup/rstat.c:71 完整函数体 */
void __css_rstat_updated(struct cgroup_subsys_state *css, int cpu)
{
	struct llist_head *lhead;
	struct css_rstat_cpu *rstatc;
	struct llist_node *self;

	/* Prevent access to uninitialized rstat pointers. */
	if (!css_uses_rstat(css))
		return;

	lockdep_assert_preemption_disabled();

	/*
	 * The lockless insertion below relies on NMI-safe cmpxchg;
	 * bail out in NMI on archs that don't provide it.
	 */
	if (!IS_ENABLED(CONFIG_ARCH_HAVE_NMI_SAFE_CMPXCHG) && in_nmi())
		return;

	rstatc = css_rstat_cpu(css, cpu);
	/*
	 * If already on list return. ... (成对内存屏障保证 flusher 可见性)
	 */
	if (llist_on_list(&rstatc->lnode))
		return;                                   // 已标脏: 直接收工 (最常见路径)

	/*
	 * This function can be renentered by irqs and nmis for the same cgroup
	 * and may try to insert the same per-cpu lnode into the llist. ...
	 * we use the fact that lnode points to itself when not on a list and
	 * then use try_cmpxchg() to atomically set to NULL to select the
	 * winner which will call llist_add(). ...
	 */
	self = &rstatc->lnode;
	if (!try_cmpxchg(&rstatc->lnode.next, &self, NULL))   // NMI/IRQ 重入安全
		return;

	lhead = ss_lhead_cpu(css->ss, cpu);                  // 该子系统在 本 cpu 的待flush链表
	llist_add(&rstatc->lnode, lhead);                    // 挂入 (无锁 llist)

	/*将该 css 及所有祖先加入 updated 树 (子脏则父必脏, flush 时用于剪枝) */
	__css_process_update_tree(css, cpu);                 // rstat.c:137
}
```

脏标记还会沿树上传（`__css_process_update_tree()` rstat.c:137，子脏则父必脏），保证 flush 时知道要扫哪些分支。

### 4.3 读取：flush 与增量聚合

读 `cpu.stat`/`memory.stat` 时触发 `css_rstat_flush()`（rstat.c:409）：

```c
/* kernel/cgroup/rstat.c:409 完整函数体 */
__bpf_kfunc void css_rstat_flush(struct cgroup_subsys_state *css)
{
	...
	for_each_possible_cpu(cpu) {
		struct cgroup_subsys_state *pos;

		__css_rstat_lock(css, cpu);
		pos = css_rstat_updated_list(css, cpu);          // 收集该 cpu 上所有脏 css
		for (; pos; pos = pos->rstat_flush_next) {
			if (is_self) {
				cgroup_base_stat_flush(pos->cgroup, cpu);   // cpu.stat 聚合
				bpf_rstat_flush(pos->cgroup, ...);       // BPF 也可以挂 flush 逻辑
			} else
				pos->ss->css_rstat_flush(pos, cpu);      // 子系统自己的聚合 (memcg)
		}
		__css_rstat_unlock(css, cpu);
		...
	}

```

聚合用的是**增量差分**（不是全量重算）：

```c
/* kernel/cgroup/rstat.c:572 完整函数体 */
static void cgroup_base_stat_flush(struct cgroup *cgrp, int cpu)
{
	struct cgroup_rstat_base_cpu *rstatbc = cgroup_rstat_base_cpu(cgrp, cpu);
	struct cgroup *parent = cgroup_parent(cgrp);
	struct cgroup_rstat_base_cpu *prstatbc;
	struct cgroup_base_stat delta;
	unsigned seq;

	/* Root-level stats are sourced from system-wide CPU stats */
	if (!parent)
		return;

	/* fetch the current per-cpu values */
	do {
		seq = __u64_stats_fetch_begin(&rstatbc->bsync);   // seqcount 读 per-cpu 副本
		delta = rstatbc->bstat;
	} while (__u64_stats_fetch_retry(&rstatbc->bsync, seq));

	/* propagate per-cpu delta to cgroup and per-cpu global statistics */
	cgroup_base_stat_sub(&delta, &rstatbc->last_bstat);   // delta = 本cpu新值 - 上次聚合值
	cgroup_base_stat_add(&cgrp->bstat, &delta);           // 累加进本组总数
	cgroup_base_stat_add(&rstatbc->last_bstat, &delta);   // 记住"已聚合到哪"
	cgroup_base_stat_add(&rstatbc->subtree_bstat, &delta);// subtree_ 视角

	/* propagate cgroup and per-cpu global delta to parent (unless that's root) */
	if (cgroup_parent(parent)) {
		delta = cgrp->bstat;
		cgroup_base_stat_sub(&delta, &cgrp->last_bstat);
		cgroup_base_stat_add(&parent->bstat, &delta);     // ↑ 上传父组
		cgroup_base_stat_add(&cgrp->last_bstat, &delta);

		delta = rstatbc->subtree_bstat;
		prstatbc = cgroup_rstat_base_cpu(parent, cpu);
		cgroup_base_stat_sub(&delta, &rstatbc->last_subtree_bstat);
		cgroup_base_stat_add(&prstatbc->subtree_bstat, &delta);
		cgroup_base_stat_add(&rstatbc->last_subtree_bstat, &delta);
	}
}
```

memcg 的 `memory.stat`/`memory.events` 走同一条路，聚合函数换成 `mem_cgroup_css_rstat_flush()`（mm/memcontrol.c:4381，把 per-cpu `vmstats_percpu` 聚合到 `memcg->vmstats`，并按 state/local/pending 三视角分别累计——对应 `memory.stat` 与 `memory.events.local`）。

### 4.4 对用户态的含义

- `cpu.stat`/`memory.stat` 读取是**有代价的**（要 flush 整棵子树的 per-CPU 副本），监控高频轮询几千个容器组本身会消耗 CPU——内核为此做了阈值触发（memcg 侧 `memcg_vmstats_needs_flush()` memcontrol.c:666，累计更新超过 `MEMCG_CHARGE_BATCH * nr_cpus` 才主动 flush 一次）；
- 数字有短暂延迟/不精确是**设计使然**，别拿 memory.stat 做强一致审计。

## 5. 7.1.3 的两个新貌

1. **rstat 从"cgroup 专属"泛化成"css 级"**：`css_rstat_*` 系列允许任何子系统给自己挂一套统计中继（`struct cgroup_subsys` 新增 `css_rstat_flush` 回调，memcg 是第一个用户）；BPF 程序也能通过 `bpf_rstat_flush()`（rstat.c:351 kfunc）参与聚合。
2. **css 创建即接 rstat 与 PSI**：`css_create()` 里 `css_rstat_init()`；cgroup 创建时 `psi_cgroup_alloc()`（psi.c:1112，由 cgroup.c:5858 调用）为该组准备压力统计——`memory.pressure` 等 PSI 文件是核心自带而非控制器提供。
