# 内核对齐工具箱：源码全景

> 源码基准: [Linux 7.1.3](../../000.SOURCE_CODE/003.LINUX-7.1.3/000.LINUX-7.1.3)（行号均 grep 实测）
>
> 书中对应原句："在 GCC 为各种 struct 或 union 组织内存布局时，会自动选择适当的对齐，使得程序员无须手工进行操作。"——编译器解决 95% 的自然对齐，内核工具箱解决剩下 5%：数值对齐、缓存行对齐、页对齐、slab 对象对齐。

## 1. 数值/地址对齐：`ALIGN` 一族

```c
// include/vdso/align.h:8
/* @a is a power of 2 value */
#define ALIGN(x, a)		__ALIGN_KERNEL((x), (a))     // 向上取整到 a 的倍数
#define ALIGN_DOWN(x, a)	__ALIGN_KERNEL((x) - ((a) - 1), (a))
#define __ALIGN_MASK(x, mask)	__ALIGN_KERNEL_MASK((x), (mask))
#define PTR_ALIGN(p, a)		((typeof(p))ALIGN((unsigned long)(p), (a)))
#define PTR_ALIGN_DOWN(p, a)	((typeof(p))ALIGN_DOWN((unsigned long)(p), (a)))
#define IS_ALIGNED(x, a)	(((x) & ((typeof(x))(a) - 1)) == 0)
```

| 宏 | 用途 | 典型调用者 |
|---|---|---|
| `ALIGN(x, a)` | 把长度/地址向上对齐（a 必须是 2 的幂） | `SKB_DATA_ALIGN`、slab 尺寸计算 |
| `ALIGN_DOWN` | 向下对齐 | 地址范围规整 |
| `IS_ALIGNED(x, a)` | 对齐断言/分支 | 调试检查、快速路径分派 |
| `PTR_ALIGN` | 指针向上对齐 | 缓冲区摆位 |

页粒度的封装（页也是一种"对齐参照系"）：

```c
// include/linux/mm.h:224
#define PAGE_ALIGN(addr) ALIGN(addr, PAGE_SIZE)
#define PAGE_ALIGN_DOWN(addr) ALIGN_DOWN(addr, PAGE_SIZE)
#define PAGE_ALIGNED(addr)	IS_ALIGNED((unsigned long)(addr), PAGE_SIZE)
```

## 2. 编译器属性：`__aligned` / `__packed`

内核把 GCC 属性包装成可移植宏：

```c
// include/linux/compiler_attributes.h:33
#define __aligned(x)                    __attribute__((__aligned__(x)))
#define __aligned_largest               __attribute__((__aligned__))
```

```c
// include/linux/compiler_attributes.h:291
#define __packed                        __attribute__((__packed__))
```

- `__aligned(N)` 用在**变量/字段/类型**上：强制最小对齐为 N（可放宽，也可**收紧**，如 `__aligned(1)` 常与 `__packed` 配对使用）；
- `__packed` 见 [001.自然对齐的定义.md](001.自然对齐的定义.md) §3——取消 padding、对齐降为 1，是"明知非对齐"的声明。

## 3. 缓存行宏：`L1_CACHE_BYTES` → `SMP_CACHE_BYTES` → `____cacheline_aligned`

编译期的来源是各架构自己的 `L1_CACHE_BYTES`：

```c
// arch/arm64/include/asm/cache.h:8     （AGI CPU / A725 / A520 同用）
#define L1_CACHE_SHIFT		(6)
#define L1_CACHE_BYTES		(1 << L1_CACHE_SHIFT)      // 64B，与 CTR_EL0 一致

// arch/x86/include/asm/cache.h:8       （EPYC 9996）
#define L1_CACHE_SHIFT	(CONFIG_X86_L1_CACHE_SHIFT)  // Kconfig.cpu:271 → X86_64 默认 6 → 64B
#define L1_CACHE_BYTES	(1 << L1_CACHE_SHIFT)
```

通用层在其上定义缓存行对齐属性（单独放 `vdso/` 是因为 vDSO 不能链内核主体）：

```c
// include/vdso/cache.h:8
#define SMP_CACHE_BYTES L1_CACHE_BYTES
#ifndef ____cacheline_aligned
#define ____cacheline_aligned __attribute__((__aligned__(SMP_CACHE_BYTES)))
#endif
```

变量级的两个变体（同时改变链接段，让对齐的数据集中摆放）：

```c
// include/linux/cache.h:71
#ifndef __cacheline_aligned
#define __cacheline_aligned					\
  __attribute__((__aligned__(SMP_CACHE_BYTES),			\
		 __section__(".data..cacheline_aligned")))
#endif
```

```c
// include/linux/cache.h:63
#ifndef ____cacheline_aligned_in_smp
#ifdef CONFIG_SMP
#define ____cacheline_aligned_in_smp ____cacheline_aligned   // SMP：按行对齐（防伪共享）
#else
#define ____cacheline_aligned_in_smp                          // UP：空宏（省内存，无伪共享问题）
#endif
#endif
```

**命名规律**：四个下划线 `____cacheline_aligned` 只加对齐属性（常用于结构体**字段**）；`__cacheline_aligned` 还把变量放进专用段（用于**全局变量**）；`_in_smp` 后缀表示仅 SMP 生效。

读多写少数据的**反向**手法（不追求对齐而追求紧凑共处一行）：

```c
// include/linux/cache.h:40
/*
 * __read_mostly is used to keep rarely changing variables out of frequently
 * updated cachelines. ... You want __read_mostly data to be tightly packed,
 * so that in the best case multiple frequently read variables for a hot path
 * will be next to each other in order to reduce the number of cachelines
 * needed to execute a critical path.
 */
#ifndef __read_mostly
#define __read_mostly
#endif
// arm64/x86 覆盖为 __section(".data..read_mostly")（arch/*/include/asm/cache.h:11 等）
```

## 4. per-CPU 数据的对齐三档

per-CPU 变量按"谁访问、多频繁"分三档对齐（`include/linux/percpu-defs.h:129`，此文件源码树中已有学习注释）：

```c
// include/linux/percpu-defs.h:129
/*
 * Declaration/definition used for per-CPU variables that must be cacheline
 * aligned under SMP conditions so that, whilst a particular instance of the
 * data corresponds to a particular CPU, inefficiencies due to direct access by
 * other CPUs are reduced by preventing the data from unnecessarily spanning
 * cachelines.
 * （每 CPU 独立变量在 SMP 下必须缓存行对齐：防止数据不必要地跨行，
 *   降低其他 CPU 直接访问时的低效 —— 详见伪共享笔记）
 */
#define DECLARE_PER_CPU_SHARED_ALIGNED(type, name)			\
	DECLARE_PER_CPU_SECTION(type, name, PER_CPU_SHARED_ALIGNED_SECTION) \
	____cacheline_aligned_in_smp
```

```c
// include/linux/percpu-defs.h:157 —— 本核独占热数据：无条件按行对齐
#define DECLARE_PER_CPU_ALIGNED(type, name)				\
	DECLARE_PER_CPU_SECTION(type, name, PER_CPU_ALIGNED_SECTION)	\
	____cacheline_aligned

// include/linux/percpu-defs.h:168 —— 页对齐档
#define DECLARE_PER_CPU_PAGE_ALIGNED(type, name)			\
	DECLARE_PER_CPU_SECTION(type, name, "..page_aligned")		\
	__aligned(PAGE_SIZE)
```

| 档 | 宏 | 语义 |
|---|---|---|
| 普通 | `DEFINE_PER_CPU` | 仅自然对齐，紧凑省内存 |
| 行对齐 | `DEFINE_PER_CPU_ALIGNED` | 本核热数据（如 runqueue），行首起步 |
| 共享但分区 | `DEFINE_PER_CPU_SHARED_ALIGNED` | 主要本核写、偶尔他核读的统计类数据，SMP 下才对齐 |

## 5. 页对齐：链接脚本里的硬保证

页表基址必须页对齐（TTBR 硬件要求），arm64 在链接脚本中直接顺序分配整页：

```
// arch/arm64/kernel/vmlinux.lds.S:235   （节选，符号=当前位置；. += PAGE_SIZE 保留整页）
	idmap_pg_dir = .;
	. += PAGE_SIZE;
#ifdef CONFIG_UNMAP_KERNEL_AT_EL0
	tramp_pg_dir = .;
	. += PAGE_SIZE;
#endif
	reserved_pg_dir = .;
	. += PAGE_SIZE;

	swapper_pg_dir = .;
	. += PAGE_SIZE;
```

链接期保证 + 构建期断言（`vmlinux.lds.S:415` 的 `ASSERT(swapper_pg_dir - reserved_pg_dir == ...)`）双层校验——对页这种 MMU 级粒度，内核不依赖运行期凑巧，而是把对齐写进布局。

## 6. slab 分配器的对齐决策

`kmem_cache_create` 用户可传 `SLAB_HWCACHE_ALIGN` 或显式 `align`，统一在 `calculate_alignment()` 收敛：

```c
// mm/slab_common.c:128
static unsigned int calculate_alignment(slab_flags_t flags,
		unsigned int align, unsigned int size)
{
	if (flags & SLAB_HWCACHE_ALIGN) {
		unsigned int ralign;

		ralign = cache_line_size();       // 运行期读 CTR_EL0 → 64B
		while (size <= ralign / 2)        // 小对象对齐减半，平衡内存浪费
			ralign /= 2;
		align = max(align, ralign);       // 用户显式要求更大对齐时尊重用户
	}

	align = max(align, arch_slab_minalign());   // 架构底线（KASAN/MTE 时为 16）
	return ALIGN(align, sizeof(void *));
}
```

三条规则一目了然：**①硬件行对齐可被用户更大的显式对齐覆盖；②小对象自动降档（1/2ⁿ 行）；③架构最低对齐是硬下限。** 架构底线在 arm64 上：

```c
// arch/arm64/include/asm/cache.h:35
#define ARCH_DMA_MINALIGN	(128)
#define ARCH_KMALLOC_MINALIGN	(8)      // kmalloc 常规对象只需 8B；
                                        // DMA 场景另走 ARCH_DMA_MINALIGN/128B
#ifdef CONFIG_KASAN_SW_TAGS
#define ARCH_SLAB_MINALIGN	(1ULL << KASAN_SHADOW_SCALE_SHIFT)  // 16B：KASAN 毒字节需要
#endif
```

（16B 这个 `ARCH_SLAB_MINALIGN` 取值与 MTE 相关：`CONFIG_KASAN_HW_TAGS` 时 `arch_slab_minalign()` 返回 `MTE_GRANULE_SIZE`——**MTE 内存标签以 16B 粒度存储**，对象不对齐到 16B 就无法独立打标签，见 `cache.h:50-53`。这是"稍大一些的对齐尺度"由硬件特性反推软件约束的又一例。运行期对外查询接口：`dma_get_cache_alignment` 即 `cache_line_size()`，`cache.h:88`。）

## 7. 实战案例：调度器 `struct rq` 的三层缓存行布局

调度器 runqueue 是"对齐工具箱"的集大成者——变量级、字段级、注释级（设计意图）同时出现：

**① 变量级：per-CPU + 行对齐**

```c
// kernel/sched/sched.h:1386
DECLARE_PER_CPU_SHARED_ALIGNED(struct rq, runqueues);
// kernel/sched/core.c:131
DEFINE_PER_CPU_SHARED_ALIGNED(struct rq, runqueues);
```

**② 字段级：把"一起用的字段"聚到专属缓存行**

```c
// kernel/sched/sched.h:1131
struct rq {
	/*
	 * The following members are loaded together, without holding the
	 * rq->lock, in an extremely hot loop in update_sg_lb_stats()
	 * (called from pick_next_task()). To reduce cache pollution from
	 * this operation, they are placed together on this dedicated cache
	 * line. Even though some of them are frequently modified, they are
	 * loaded much more frequently than they are stored.
	 */
	unsigned int		nr_running;
	...
	unsigned long		cpu_capacity;
```

```c
// kernel/sched/sched.h:1163 / 1220 —— 字段级 ____cacheline_aligned：显式开新行
	u64			nr_switches	____cacheline_aligned;
	...
	u64			clock_task ____cacheline_aligned;

// kernel/sched/sched.h:493 —— cfs_rq.load_avg 同样按行分隔
	atomic_long_t		load_avg ____cacheline_aligned;
```

**③ 设计意图（注释原文）**："placed together on this **dedicated cache line** …to reduce **cache pollution**"——字段分组的标准做法：读聚一行（负载均衡的热读字段）、写隔离行（频繁写的计数器各自开行，防伪共享）。读 [003.非对齐访问-高速缓存性能视角.md](003.非对齐访问-高速缓存性能视角.md) §2-§5 理解每一行的物理动机。
